\section{Lec2: Advanced Ensemble Learning}

\subsection{Gradient Boosting:}
Boosting is an ensemble technique that combines multiple weak learners to create a strong learner. The main idea is to train models sequentially, with each model focusing on the errors made by the previous ones.
(用人话说，就是一波接一波地训练模型，每一波都专注于纠正前一波的错误，从而逐步提升整体的预测能力。)

\begin{itemize}
    \item \textbf{Final Model:} \[H_T(\mathbf{x}) = \sum_{t=1}^{T} \alpha_t h_t(\mathbf{x})\]
    
    \begin{itemize}
        \item The weighted sum of all weak learners(加权和).
        \item $H_T(\mathbf{x})$ is the final model after $T$ rounds.
        \item $h_t(\mathbf{x})$ is the $t$-th weak learner.
        \item $\alpha_t$ is the weight of the $t$-th weak learner.
    \end{itemize}
    
    \item \textbf{Loss Function:} \[\mathcal{L}(H) := \frac{1}{n} \sum_{i=1}^{n} \ell(H(\mathbf{x}_i), y_i)\]

    \begin{itemize}
        \item \textbf{Each Step:} We want to add a function $h$ to minimize the loss as fast as possible. Using first-order Taylor expansion(一阶泰勒展开):
        \[
        \mathcal{L}(H + \alpha h) \approx \mathcal{L}(H) + \alpha \langle \nabla \mathcal{L}(H), h \rangle
        \]
        
        \item \textbf{Find $h$:} Minimize $\langle \nabla \mathcal{L}(H), h \rangle$, i.e.:
        \[
        h = \arg\min_{h \in \mathbb{H}} \sum_{i=1}^{n} \frac{\partial \mathcal{L}}{\partial [H(\mathbf{x}_i)]} h(\mathbf{x}_i)
        \]
        
        \begin{itemize}
            \item Here $\frac{\partial \mathcal{L}}{\partial [H(\mathbf{x}_i)]}$ is the gradient of the loss function w.r.t. the current model's output on the $i$-th sample.
            \item We train $h$ to fit these \textbf{negative gradients(负梯度)}. 
        \end{itemize}
       
        
    \end{itemize}
     \item \textbf{GBR with squared error loss: } For regression tasks with squared error loss:
     \[\ell(H(\mathbf{x}_i), y_i) = \sum_{i=1}^n (y_i - H(\mathbf{x}_i))^2\]
        \begin{itemize}
            \item Solve $h_{t+1} = \arg\min_{h \in \mathbb{H}} \sum_{i=1}^{n} q_i h(\mathbf{x}_i)$, where $q_i = \frac{\partial \mathcal{L}}{\partial [H(\mathbf{x}_i)]}$
            
            \item Let $\sum_{i=1}^{n} h^2(\mathbf{x}_i) = \text{constant}$ (we can always normalize the predictions(对结果归一化处理)) and replace $q_i$ with $-2r_i$. We have
            \begin{align}
                h_{t+1} &= \arg\min_{h \in \mathbb{H}} \sum_{i=1}^{n} q_i h(\mathbf{x}_i) \notag \\
                &= \arg\min_{h \in \mathbb{H}} -2 \sum_{i=1}^{n} r_i h(\mathbf{x}_i) \notag \\
                &= \arg\min_{h \in \mathbb{H}} \sum_{i=1}^{n} \left( r_i^2 - 2r_i h(\mathbf{x}_i) + (h(\mathbf{x}_i))^2 \right) \\
                &= \arg\min_{h \in \mathbb{H}} \sum_{i=1}^{n} (h(\mathbf{x}_i) - r_i)^2 \notag
            \end{align}
            
            \item We train $h_{t+1}$ to predict $r_i$, which are from the old model $H_t$.
            
            \item The gradient is:
            \[\frac{\partial \mathcal{L}}{\partial [H(\mathbf{x}_i)]} = -2(y_i - H(\mathbf{x}_i))\]
            
            \item So we fit $h$ to the residuals:
            \[r_i = y_i - H(\mathbf{x}_i)\]
            
            \item Update the model:
            \[H_{t}(\mathbf{x}) = H_{t-1}(\mathbf{x}) + \alpha_t h_t(\mathbf{x})\]
        \end{itemize}
    \item \textbf{GBR with Absolute loss (更鲁棒): } For regression tasks with absolute loss:
        \begin{itemize}
            \item Square loss is easy to deal with mathematically but not robust to outliers.
            \item Absolute loss (more robust to outliers):
            \[\ell(y, \hat{y}) = |y - \hat{y}|\]
        \end{itemize}
        
        \begin{itemize}
            \item The gradient is $q_i = \frac{\partial \mathcal{L}}{\partial H(\mathbf{x}_i)} = -\text{sign}(y_i - H(\mathbf{x}_i))$.
            \item Then fit $h$ on $-q_i$, $i = 1, 2, \ldots, n$. (no longer the residuals, different from using the squared loss)
        \end{itemize}
    \item \textbf{GBR with Huber loss(更更鲁棒): } 
        \begin{itemize}
            \item Huber loss (more robust to outliers):
            \[
            \ell(y, \hat{y}) = \begin{cases}
                \frac{1}{2}(y - \hat{y})^2 & |y - \hat{y}| \leq \delta \\
                \delta(|y - \hat{y}| - \delta/2) & |y - \hat{y}| > \delta
            \end{cases}
            \]
            
            \item The gradient is:
            \[
            \frac{\partial \mathcal{L}}{\partial H(\mathbf{x}_i)} = \begin{cases}
                -(y_i - H(\mathbf{x}_i)) & |y_i - H(\mathbf{x}_i)| \leq \delta \\
                -\delta \, \text{sign}(y_i - H(\mathbf{x}_i)) & |y_i - H(\mathbf{x}_i)| > \delta
            \end{cases}
            \]
        \end{itemize}
    \item \textbf{GBM for classification: } 
        \begin{itemize}
            \item Predict probability of $K$ classes:
            \[p_k(\mathbf{x}) = \frac{\exp(h^{(k)}(\mathbf{x}))}{\sum_{c=1}^{K} \exp(h^{(c)}(\mathbf{x}))} \triangleq \hat{y}^{(k)}, \quad k = 1, 2, \ldots, K\]
            
            \item Loss: $\mathcal{L}(H) = \sum_{i=1}^{n} \ell(\mathbf{y}_i, \hat{\mathbf{y}}_i)$ (e.g. cross-entropy or KL divergence)
            
            \item Initialize $H^{(1)}, H^{(2)}, \ldots, H^{(K)}$, iterate until converge or reach max $T$:
            \begin{enumerate}
                \item Calculate negative gradients for every class:
                \[-g_k(\mathbf{x}_i) = -\frac{\partial \mathcal{L}}{\partial [H^{(k)}(\mathbf{x}_i)]}, \quad i = 1, \ldots, n, \ k = 1, \ldots, K\]
                
                \item Fit $h^{(k)}$ to $-g_k(\mathbf{x}_i)$ (负梯度), $k = 1, 2, \ldots, K$.
                
                \item Update: $H^{(k)} \leftarrow H^{(k)} + \alpha h^{(k)}$, $k = 1, 2, \ldots, K$.
            \end{enumerate}
        \end{itemize}
\end{itemize}

\subsection{AdaBoost}
A special case of gradient boosting with exponential loss.

\begin{itemize}
    \item Exponential loss (learns $\alpha$ adaptively):
    \[\mathcal{L}(H) = \sum_{i=1}^{n} e^{-y_i H(\mathbf{x}_i)}\]
    
    \item Gradient: $q_i = \frac{\partial \mathcal{L}}{\partial H(\mathbf{x}_i)} = -y_i e^{-y_i H(\mathbf{x}_i)}$
    
    \item Let $w_i = \frac{1}{Z} e^{-y_i H(\mathbf{x}_i)}$, where $Z = \sum_{i=1}^{n} e^{-y_i H(\mathbf{x}_i)}$ (constant w.r.t $h$), so $\sum_{i=1}^{n} w_i = 1$.
    
    $w_i$ is the relative contribution of $(\mathbf{x}_i, y_i)$ to the overall loss.
    
    \item Binary classification: $y \in \{-1, +1\}$, $h(\mathbf{x}) \in \{-1, +1\}$.
    
    \item Derivation:
    \begin{align}
        h &= \arg\min_{h \in \mathbb{H}} \sum_{i=1}^{n} q_i h(\mathbf{x}_i) \notag \\
        &= \arg\min_{h \in \mathbb{H}} -\sum_{i=1}^{n} w_i y_i h(\mathbf{x}_i) \notag \\
        &= \arg\min_{h \in \mathbb{H}} \sum_{i: h(\mathbf{x}_i) \neq y_i} w_i - \sum_{i: h(\mathbf{x}_i) = y_i} w_i \notag \\
        &= \arg\min_{h \in \mathbb{H}} \sum_{i: h(\mathbf{x}_i) \neq y_i} w_i \notag
    \end{align}
    * Last equality holds because $\sum_{i=1}^{n} w_i = 1$.
    
    \item Result: 
    \[h = \arg\min_{h \in \mathbb{H}} \sum_{i: h(\mathbf{x}_i) \neq y_i} w_i\]
    
    \[\epsilon := \sum_{i: h(\mathbf{x}_i) \neq y_i} w_i\] is the weighted classification error(加权错误率).
    
    Note: misclassified points by $H$ get larger weights(分类错误的点会得到更大的权重).

    \item Given $h$, find $\alpha$ via:
    \[\alpha = \arg\min_{\alpha} \mathcal{L}(H + \alpha h) = \arg\min_{\alpha} \sum_{i=1}^{n} e^{-y_i(H(\mathbf{x}_i) + \alpha h(\mathbf{x}_i))}\]
    
    \item Differentiate w.r.t $\alpha$ and set to zero:
    \[\sum_{i=1}^{n} y_i h(\mathbf{x}_i) e^{-(y_i H(\mathbf{x}_i) + \alpha y_i h(\mathbf{x}_i))} = 0\]
    
    It follows that:
    \[\sum_{i: h(\mathbf{x}_i) y_i = 1} e^{-(y_i H(\mathbf{x}_i) + \alpha y_i h(\mathbf{x}_i))} - \sum_{i: h(\mathbf{x}_i) y_i = -1} e^{-(y_i H(\mathbf{x}_i) + \alpha y_i h(\mathbf{x}_i))} = 0\]
    \[\sum_{i: h(\mathbf{x}_i) y_i = 1} w_i e^{-\alpha} - \sum_{i: h(\mathbf{x}_i) y_i = -1} w_i e^{\alpha} = 0\]
    
    We have $(1 - \epsilon) e^{-\alpha} - \epsilon e^{\alpha} = 0$, $e^{2\alpha} = \frac{1-\epsilon}{\epsilon}$, and get:
    \[\boxed{\bm{\alpha = \frac{1}{2} \ln \frac{1 - \epsilon}{\epsilon}}}\]
    
\end{itemize}

\subsection{Mixture of Experts(MoE)}
A machine learning technique where multiple expert learners 
(e.g. neural networks) are used to divide a problem space into
homogeneous regions (distinct subtasks).
(用人话(AI)说就是，把一个复杂的问题拆分成多个子任务，每个子任务由一个专家模型来处理，从而提升整体的学习效果。)\\
(骗你的人话也没看懂。。)

    \subsubsection{The First Attempt}
\begin{itemize}
    \item Error function:
    \[E = \left\| y - \sum_{j=1}^{k} g_j O_j \right\|^2\]
    
    \item $y$: target vector; $O_j$: output of expert $j$; $g_j$: proportional contribution of expert $j$.
    
    \item *This error function does not ensure localisation of experts.(人话：专家没有明确的分工).
\end{itemize}

\subsubsection{The Second Attempt}
\begin{itemize}
    \item Error function:
    \[E = \sum_{j=1}^{k} g_j \| y - O_j \|^2\]
    
    \item The system tends to devote a single expert to each training case.
    
    \item *This may not work well in practice.(人话：实际效果可能不佳)
\end{itemize}

\subsubsection{The Third Attempt [Jacobs et al. 1991]}
\begin{itemize}
    \item Error function (mixture of Gaussians):
    \[E_{ME} = -\log \sum_{j=1}^{k} g_j \exp\left( -\frac{1}{2} (y - O_j)^T \Sigma^{-1} (y - O_j) \right)\]
    
    \item Assume $\Sigma = I$ ($\Sigma$: Covariance matrix, 协方差矩阵), derivative w.r.t the $i$-th expert:
    \[\frac{\partial E_{ME}}{\partial O_i} = -\left[ \frac{g_i \exp\left( -\frac{1}{2} (y - O_i)^T (y - O_i) \right)}{\sum_j g_j \exp\left( -\frac{1}{2} (y - O_j)^T (y - O_j) \right)} \right] (y - O_i)\]
    
    \item Compare with derivative of second attempt:
    \[\frac{\partial E}{\partial O_i} = -2g_i (y - O_i)\]
    
    \item The former considers how well expert $i$ performs relative to others, adapting the best-fitting expert faster.
    
    E.g., $g_1 = 0.8$, $g_2 = 0.2$, then $\frac{0.8 \times 0.9}{0.8 \times 0.9 + 0.2 \times 0.1} \approx 0.97 > 0.8$.\\
    (人话：表现好的专家会得到更快的提升)
\end{itemize}
(这nm都是什么玩意?)

\subsection{Stacking(堆叠法)}
Multiple base learners' outputs $(\hat{y}_1, \hat{y}_2, \ldots, \hat{y}_N)$ are combined by a meta-learner (stacker) to produce the final prediction $\hat{Y}$.
(用人话说就是，把多个模型的预测结果再拿去训练一个新的模型，从而提升整体的预测效果。)
\begin{itemize}
    \item Multi-level stacking: stacking can be repeated.
    \item Popular stackers: linear models (fast), gradient boosting (accurate).
    \item Base models should be diverse, expert at different data parts.
    \item Trade-off: accurate but slow to predict.
\end{itemize}
(为什么一讲能有这么多神秘小知识点和神秘公式？)

\section{Lec3: Advanced Applications}

\subsection{Recommendation Systems}

\subsubsection{Collaborative Filtering(协同过滤)}
Core idea: Use behavioral data from many users (e.g., ratings, clicks) to predict what the current user might like.
(人话：大数据推荐你喜欢的东西)

\begin{itemize}
    \item \textbf{User-Item Interaction:}
    \begin{itemize}
        \item Explicit Feedback(显式反馈): ratings, purchases.
        \item Implicit Feedback(隐式反馈): clicks, browsing time.
    \end{itemize}
    \item \textbf{User-Item Rating Matrix:} Rows = users, columns = items, values = ratings. Typically large and sparse.
    \item \textbf{Distance/Similarity Measurement(相似度度量):}
    \begin{itemize}
        \item Euclidean distance: $\text{sim}(user_i, user_j) = \frac{1}{1 + \|\mathbf{x}_i - \mathbf{x}_j\|_2}$
        \item Cosine similarity: $\text{sim}(user_i, user_j) = \frac{\mathbf{x}_i \cdot \mathbf{x}_j}{\|\mathbf{x}_i\| \|\mathbf{x}_j\|}$
        \item Pearson correlation: $\rho_{X,Y} = \frac{\text{cov}(X, Y)}{\sigma_X \sigma_Y}$
    \end{itemize}

    \item \textbf{Nearest-Neighbor Collaborative Filtering(最近邻协同过滤):}
    
    Predict utility of item $i$ based on similar users who rated that item.
    \begin{itemize}
        \item $\mathcal{N}$: neighborhood set (most similar users to $u$ who rated item $i$)(人话：给你推荐东西的那些“邻居”用户)
        \item $w_{uv} \in [0,1]$: similarity weight between users $u$ and $v$(人话：你和邻居用户的相似度)
        \item Prediction:
        \[\hat{x}_{ui} = \bar{x}_u + \sum_{v \in \mathcal{N}} \left( (x_{vi} - \bar{x}_v) \times \frac{w_{uv}}{\sum_{v' \in \mathcal{N}} w_{uv'}} \right)\]
        \item $\bar{x}_u = \frac{1}{|I_u|} \sum_{i \in I_u} x_{ui}$ (average rating of user $u$, where $I_u$ is the set of items rated by $u$)
        \item $\bar{x}_v = \frac{1}{|I_v|} \sum_{i \in I_v} x_{vi}$ (average rating of user $v$, where $I_v$ is the set of items rated by $v$)
    \end{itemize}

    \item \textbf{Matrix Factorization Collaborative Filtering(矩阵分解协同过滤):}
    \begin{itemize}
        \item Notations: 
        \begin{itemize}
            \item $R = [r_{ui}] \in \mathbb{R}^{m \times n}$: incomplete user-item rating matrix
            \item $\Omega$: set of observed entries (known ratings)
            \item $P = [p_1, \ldots, p_u, \ldots, p_m] \in \mathbb{R}^{f \times m}$, \quad $Q = [q_1, \ldots, q_i, \ldots, q_n] \in \mathbb{R}^{f \times n}$
        \end{itemize}
        
        \item Basic SVD ($R \approx P^T Q$):
        \[\min_{P,Q} \sum_{(u,i) \in \Omega} \left\{ (r_{ui} - p_u^T q_i)^2 + \lambda (\|p_u\|^2 + \|q_i\|^2) \right\}\]
        
        \item SVD with bias ($b_{ui} = \mu + b_u + b_i$):
        \[\min_{P,Q,B} \sum_{(u,i) \in \Omega} \left\{ (r_{ui} - \mu - b_u - b_i - p_u^T q_i)^2 + \lambda (\|p_u\|^2 + \|q_i\|^2 + b_u^2 + b_i^2) \right\}\]
        
        \item $\mu$: global mean; $b_u$: user bias; $b_i$: item bias; 
        \item $\lambda(\cdot)$: regularization term (prevents overfitting).
        \item Optimization: GD/SGD or Alternating Least Squares(交替最小二乘法).
    \end{itemize}

    \item \textbf{Pros \& Cons of Collaborative Filtering:}
    \begin{itemize}
        \item Pros: No domain knowledge needed; captures diverse user preferences(人话：不需要领域知识，能捕捉多样的用户偏好).
        \item Cons: Cold-start problem (new users/items); data sparsity(人话：新来的用户和物品数据较少，导致推荐效果差).
    \end{itemize}
\end{itemize}

\subsubsection{Content-Based Methods}
    \begin{itemize}
        \item \textbf{Content analysis:} item $\to$ feature vector $v$ (e.g. TF-IDF, image features).
        \item \textbf{Profile learning:} user $\to$ feature vector $z$ (e.g. age, sex, education).
        \item \textbf{Filtering module:} train classification/regression model to predict user's utility for an item.
        \item \textbf{Recommendation for user:}
        \begin{itemize}
            \item $n$: \# of items; 
            \item $z_u \in \mathbb{R}^d$: 
            \item user $u$'s feature vector; 
            \item $h: \mathbb{R}^d \to \mathbb{R}^n$ (e.g. neural network);
            \item $h_i$: $i$-th output of $h$.
            \item $\ell$: loss function (e.g. squared loss).
            \[\min_h \sum_{(u,i) \in \Omega} \ell(r_{ui}, h_i(z_u))\]
        \end{itemize}

        \item \textbf{Recommendation for item:}
        \begin{itemize}
            \item $m$: \# of users; 
            \item $v_i \in \mathbb{R}^{d'}$: item 
            \item $i$'s feature vector; 
            \item $g: \mathbb{R}^{d'} \to \mathbb{R}^m$ (e.g. neural network); 
            \item $g_u$: $u$-th output of $g$.
            \item $\ell$: loss function (e.g. squared loss).
            \[\min_g \sum_{(u,i) \in \Omega} \ell(r_{ui}, g_u(v_i))\]
        \end{itemize}
        \item \textbf{Pros \& Cons of Content-Based Methods:}
        \begin{itemize}
            \item Pros: User-independent; explainable; handles new items/users well.
            \item Cons: Needs domain knowledge; narrow recommendations (similar items).
        \end{itemize}
    \end{itemize}

\subsubsection{Hybrid Methods(看起来是不重要的知识点)}
Most modern systems are hybrid recommenders.
    \begin{itemize}
        \item Combine separate recommenders (CF + CB): ensemble techniques (linear weighting, stacking, etc.)
        \item Add content-based aspects to CF: e.g. matrix factorization with side information.
    \end{itemize}

\subsubsection{Evaluation Metrics for RS}
   \paragraph{3.1.4.1 Prediction Metrics(评分预测指标)}
    \begin{itemize}
        \item \textbf{Mean Absolute Error (MAE):}
        \[\text{MAE} = \frac{1}{|\mathcal{T}|} \sum_{(u,i) \in \mathcal{T}} |r_{ui} - \hat{r}_{ui}|\]
        
        \item \textbf{Root Mean Squared Error (RMSE):}
        \[\text{RMSE} = \sqrt{\frac{1}{|\mathcal{T}|} \sum_{(u,i) \in \mathcal{T}} (r_{ui} - \hat{r}_{ui})^2}\]
        
        Here $\mathcal{T}$ denotes the set of user--item pairs used for evaluation (e.g., the test set).(人话：测试集中所有用户和物品的组合)
    \end{itemize}
    \paragraph{3.1.4.2 Ranking-based Metrics(基于排序的指标)}
    \begin{itemize}
        \item \textbf{Precision@K:} fraction of top-$K$ recommended items that are relevant(人话：前K个推荐中有多少是用户喜欢(相关的)).
        \[\text{Prec}(R)_k = \frac{|\{r \in R:\ r \le k\}|}{k}\]

        \item \textbf{Recall@K:} fraction of relevant items covered in top-$K$(人话：前K个推荐中，用户喜欢的相关物品占用户喜欢的物品总数).
        \[\text{Recall}(R)_k = \frac{|\{r \in R:\ r \le k\}|}{|R|}\]
        \begin{itemize}
            \item (Precision = TP/(TP+FP); Recall = TP/(TP+FN)).
            \item $r$ = rank position of a recommended item; 
            \item $k$ = cut-off (top-$k$); 
            \item $R$ = set of relevant items for the user.
        
        \end{itemize}
        \item \textbf{Average Precision (AP@N):} average of precision values at ranks of relevant items.(前N个推荐中相关物品的精确率)
        \[\text{AP@}N = \frac{1}{m} \sum_{k=1}^{N} P(k) \cdot \text{rel}(k)\]
        where $P(k)$ is precision@k, $m$ number of relevant items, and $\text{rel}(k)$ is indicator if item at rank $k$ is relevant.

        \item \textbf{Mean Average Precision (MAP):} mean of AP over $Q$ users: 
        \[\text{MAP} = \frac{1}{Q} \sum_{q=1}^{Q} \text{AP}(q)\]

        \item \textbf{Normalized Discounted Cumulative Gain (NDCG):} evaluates ranked relevance with position discounting.
        \[\text{NDCG}_p = \frac{\text{DCG}_p}{\text{IDCG}_p},\quad \text{DCG}_p = \sum_{i=1}^{p} \frac{2^{\text{rel}_i}-1}{\log_2(i+1)}\]
        where $\text{rel}_i$ is relevance of item at rank $i$, and $\text{IDCG}_p$ is the ideal DCG (sorted by relevance). \\
        Range: $[0,1]$. (NDCG越接近1越好)

        \begin{itemize}
            \item \textbf{Example:} 5 recommended items with relevances $[3,2,1,0,2]$ (in rank order).
            \begin{align*}
            \text{DCG}_5 &= \frac{2^{3}-1}{\log_2(1+1)} + \frac{2^{2}-1}{\log_2(2+1)} + \frac{2^{1}-1}{\log_2(3+1)} + \frac{2^{0}-1}{\log_2(4+1)} + \frac{2^{2}-1}{\log_2(5+1)} \approx 10.5538,\\
            \text{IDCG}_5 &= \text{DCG}_5(\text{sorted rel}=[3,2,2,1,0]) \approx 10.8235,\\
            \text{NDCG}_5 &= \dfrac{\text{DCG}_5}{\text{IDCG}_5} \approx 0.975.
            人话说 IDCG 是理想情况下的 DCG（相关性排序），就是 DCG 计算的时候按照相关性降序排序计算
            \end{align*}
        \end{itemize}

\section{Lec4-1: Graph Cut and Spectral Clustering(谱聚类)}
\subsection{Graph Paritition}
A similarity graph G=(V,E,W) represents data points as vertices V, 
with an edge in E when the pairwise similarity is positive and weights W storing those affinities.
The affinity matrix records these pairwise similarities.
Graph partitioning (clustering) aims to split the graph so that edges inside a group have large weights while edges across groups have small weights.
(人话：图划分就是把图分成若干部分，使得每个部分内的节点之间联系紧密(组内权重大)，而不同部分之间的联系较弱(组间权重小)。)
\\

Given data points, a similarity graph can be constructed using methods such as \textbf{k-nearest neighbor or $\epsilon$-neighborhood}. The edge weights are often defined by a \textbf{Gaussian kernel}:
\[
k(x_i, x_j) = \exp\left(-\frac{\|x_i - x_j\|^2}{2\sigma^2}\right)
\]

\subsection{Minimum Cut}
\begin{itemize}
    \item Minimum cut: partition the graph into two sets $A$ and $B$ minimizing
    $\mathrm{cut}(A,B):=\sum_{i\in A,\,j\in B} w_{ij}$.
    \item Solvable efficiently (e.g. via max-flow/min-cut algorithms, typical cost O(|V||E|)),
    but the minimum cut often yields unbalanced solutions (it may isolate vertices).
    (人话：最小割可以高效求解，但结果往往不平衡，可能会把一些节点孤立出来。)

        \label{fig:minimum-cut}
    \end{figure}
    (画的不错的一张图)
    \item To address this, we can use \textbf{Normalized Cut (Ncut)}:
\end{itemize}

\subsection{Normalized Cut}
% Concise summary of Normalized Cut
Normalized Cut balances cut weight with cluster sizes. For a partition (A,B), define
\[\mathrm{vol}(A)=\sum_{i\in A} d_i,\quad d_i=\sum_{j} w_{ij},\]
and
\[\mathrm{Ncut}(A,B):=\mathrm{cut}(A,B)\left(\frac{1}{\mathrm{vol}(A)}+\frac{1}{\mathrm{vol}(B)}\right).
\]
\begin{itemize}
    \item Minimizing Ncut favors balanced partitions but is \textbf{NP-hard};
    \item \textbf{spectral clustering (谱聚类)} provides an efficient relaxation.
\end{itemize}
 
\subsubsection{Degree Matrix and Graph Laplacian}

\[
W=\begin{bmatrix}
w_{11} & w_{12} & \cdots & w_{1N} \\
w_{21} & w_{22} & \cdots & w_{2N} \\
\vdots & \vdots & \ddots & \vdots \\
w_{N1} & w_{N2} & \cdots & w_{NN}
\end{bmatrix}
\]

\[
D=\begin{bmatrix}
d_1 & 0 & \cdots & 0 \\
0 & d_2 & \cdots & 0 \\
\vdots & \vdots & \ddots & \vdots \\
0 & 0 & \cdots & d_N
\end{bmatrix},\qquad d_j=\sum_{i=1}^N w_{ij}.
\]

\begin{itemize}
    \item $D$ is the \textbf{degree matrix}. And the \textbf{graph Laplacian matrix} is defined as
    \[L:=D-W.
    \]
    \item Properties: $L$ is symmetric positive semi-definite, $\mathbf{1}^\top L=0$, and its eigenvectors are used in spectral clustering (relaxation of Ncut).
\end{itemize}

\subsubsection{Normalized Cut and Graph Laplacian}

\subsubsection*{Mathematical derivation (optional)}

Recall $L=D-W$ and $D=\mathrm{diag}(d_1,\ldots,d_N)$. 

Let $u=[u_1,u_2,\ldots,u_N]^\top$ with
\[
u_i=\begin{cases}
\dfrac{1}{\mathrm{vol}(A)}, &\text{if } i\in A,\\[4pt]
-\dfrac{1}{\mathrm{vol}(B)}, &\text{if } i\in B.
\end{cases}
\]

Then
\[
u^\top L u=\tfrac{1}{2}\sum_{i,j} w_{ij}(u_i-u_j)^2=\sum_{i\in A, j\in B} w_{ij}\left(\frac{1}{\mathrm{vol}(A)}+\frac{1}{\mathrm{vol}(B)}\right)
\]
and
\[
u^\top D u=\sum_i d_i u_i^2=\sum_{i\in A}\frac{d_i}{\mathrm{vol}(A)^2}+\sum_{j\in B}\frac{d_j}{\mathrm{vol}(B)^2}=\frac{1}{\mathrm{vol}(A)}+\frac{1}{\mathrm{vol}(B)}.
\]

Therefore
\[
\frac{u^\top L u}{u^\top D u}=\sum_{i\in A, j\in B} w_{ij}\left(\frac{1}{\mathrm{vol}(A)}+\frac{1}{\mathrm{vol}(B)}\right)=\mathrm{Ncut}(A,B).
\]

\subsubsection*{Conclusion}
\begin{itemize}
    \item Ncut is equivalent to minimizing the Rayleigh quotient $\dfrac{u^\top L u}{u^\top D u}$, i.e.,
    \[
    \min_{A,B}\mathrm{Ncut}(A,B)\iff \min_{u}\frac{u^\top L u}{u^\top D u},\quad u\in\mathbb{R}^N,\quad
    u_i=\begin{cases}\dfrac{1}{\mathrm{vol}(A)},&i\in A,\\[4pt]-\dfrac{1}{\mathrm{vol}(B)},&i\in B.\end{cases}
    \]
    \item Equivalent formulation: minimize the quotient subject to $u^\top D\mathbf{1}=0$ and binary constraints $u_i\in\{1,-b\}$ (for some $b>0$).
    \item Relaxation: 
    \[
    L u = \lambda D u,
    \]
    taking the eigenvector corresponding to the \textbf{second smallest eigenvalue} as the relaxed solution.
    \item Equivalently, use the normalized Laplacian $\widetilde{L}=D^{-1}L=I-D^{-1}W$.\\ Obtain a binary partition by thresholding $u$ at $0$: $i\in A$ if $u_i\ge0$, else $i\in B$.
    \item Extend to $k$ clusters by using the first $k$ nontrivial eigenvectors and applying $k$-means (spectral clustering).
    \item (人话：2类划分找第二小特征值对应的特征向量，直接用正负号判断。而k类划分用前k个非平凡特征向量，然后要多一步k-means聚类，不能直接简单的判断符号。)
\end{itemize}

\subsection{Spectral Clustering Algorithm}
\begin{itemize}
    \item Input: data $X=\{x_1,x_2,\ldots,x_N\}$ and number $K$ of clusters.
    \item \textbf{Step 1}: Construct a similarity (affinity) matrix $W$ (e.g. Gaussian kernel)
    \[w_{ij}=\exp\left(-\frac{\|x_i-x_j\|^2}{2\sigma^2}\right)\]
    and build either a $k$-nearest neighbor or $\epsilon$-neighborhood graph.
    \item \textbf{Step 2}: Compute Laplacian $L$ (or normalized variants):
    \[L=D-W,\qquad \widetilde{L}=D^{-1}L=I-D^{-1}W,\qquad \hat{L}_{sym}=I-D^{-1/2}WD^{-1/2}.
    \]
    \item \textbf{Step 3}: Eigen-decompose (normalized) Laplacian and take first $K$ nontrivial eigenvectors to form $Z= [v_1,\ldots,v_K]^\top\in\mathbb{R}^{K\times N}$.(对 L 做特征值分解)
    \item \textbf{Step 4}: Normalize the columns (row-wise embedding) to unit $\ell_2$ norm(归一化):
    \[z_i\leftarrow z_i/\|z_i\|,\quad i=1,\ldots,N.
    \]
    \item \textbf{Step 5}: Run $K$-means on $\{z_1,\ldots,z_N\}$ and output $K$ clusters (assignments on $Z$ or map back to $X$).
    \item Notes: use $\widetilde{L}_{sym}$ (symmetric normalized) for best numerical stability; thresholding the second eigenvector recovers a binary partition.
    \item \textbf{Properties of L:}\\
    For $L = D - W$ or $\hat{L} = I - D^{-1/2} W D^{-1/2}$
    \begin{itemize}
        \item $L$ (and $\hat{L}$) are symmetric and positive semi-definite.(对称 + 半正定)
        \item Eigenvalues satisfy $0=\lambda_1\le\lambda_2\le\cdots\le\lambda_N$.
        \item The multiplicity $K$ of eigenvalue $0$ equals the number of connected components of the graph (hence $K$ clusters).
        (0特征值的数量等于图的连通分量数，也就是聚类数)
    \end{itemize}
\end{itemize}

\section{Lec4-2: Semi-Supervised Learning (SSL, 半监督学习)}
\subsection{Notations}

\begin{itemize}
    \item Input (or feature) $\mathbf{x} \in \mathcal{X}$, output (or label) $\mathbf{y} \in \mathcal{Y}$
    \item Learner $f : \mathcal{X} \to \mathcal{Y}$
    \item Labeled data $(X_l, Y_l) = \{(\mathbf{x}_1, \mathbf{y}_1), \dots, (\mathbf{x}_l, \mathbf{y}_l)\}$
    \item Unlabeled data $X_u = \{\mathbf{x}_{l+1}, \dots, \mathbf{x}_N\}$, available during training
    \item Loss function $\ell : \mathcal{Y} \times \mathcal{Y} \to \mathbb{R}$
    \item Usually, $l \ll N$
    \item Test data $X_{\text{test}} = \{ \mathbf{x}_{N+1}, \dots \}$, not available during training
\end{itemize}
\paragraph{Why Semi-Supervised Learning?} ~\\
Labeled data is often \textbf{scarce, expensive}, and requires expert annotation or specialized equipment. 
In contrast, unlabeled data is \textbf{abundant and cost-effective}. 
By leveraging both, we can build more robust and high-performing machine learning models.

\subsection{Self-Training Algorithm}
% --- 第一个 subsubsection：自训练算法基础 ---
\paragraph{Assumption:} One's own high confidence predictions are correct.(人话：自己高置信度的预测是正确的)

\subsubsection{Algorithm Procedure:}
\begin{enumerate}
    \item Train model $f$ from labeled data $(X_l, Y_l)$.
    \item Predict labels for unlabeled data $x \in X_u$. (用有标签数据训练模型，然后对无标签数据进行预测)
    \item Add the pseudo-labeled pairs $(x, f(x))$ to the labeled training set. (把预测结果当作新的标签加入训练集)
    \item Repeat the process until a stopping criterion is met.
\end{enumerate}

% --- 第二个 subsubsection：算法变体 ---
\subsubsection{Variations of Self-Training}
Depending on how pseudo-labeled data is incorporated, there are several variations:
\begin{itemize}
    \item \textbf{Most Confident:} Add only a few samples $(x, f(x))$ with the highest prediction confidence to the labeled data. (人话：只添加那些模型最有信心的预测结果)
    \item \textbf{All Samples:} Add all predicted $(x, f(x))$ directly to the labeled data.(直接把所有预测结果都加入训练集)
    \item \textbf{Weighted Approach:} Add all $(x, f(x))$ to the labeled data, but assign different weights to each sample according to its prediction confidence.
    (根据预测置信度给每个样本分配不同的权重)
\end{itemize}

% --- 第三个 subsubsection：具体应用实例 ---
\subsubsection{Self-Training Algorithm: Propagating 1-NN (传播1-NN)}
This is a specific instance of self-training using the 1-Nearest Neighbor (1-NN) approach:
\begin{enumerate}
    \item Classify an unlabeled sample $x$ using the 1-NN rule (assigning the label of its nearest labeled neighbor).(用1-NN方法对无标签样本进行分类，分配最近的有标签邻居的标签)
    \item Add the newly labeled pair $(x, f(x))$ to the labeled dataset and repeat the process.
\end{enumerate}

\paragraph{Advantages and Disadvantages} 
\begin{itemize}
    \item \textbf{Pros:} It is the simplest semi-supervised approach, serves as a versatile wrapper method, and is widely utilized in Natural Language Processing (NLP).
    \item \textbf{Cons:} The method is sensitive to outliers, and early classification errors can lead to self-reinforcing mistakes.
    (对异常值敏感)
\end{itemize}

\subsection{Graph based SSL methods}
\paragraph{Assumption: }A graph is given on the labeled and unlabeled data.  
Instances connected by heavy edge tend to have the same label(由重边连接的实例往往具有相同的标签)
\\
人话就是边的权重反应了两个节点(数据点)，之间的相似程度，边权越大，数据点越相似
\subsubsection{Graph Construction} 
\begin{itemize}
    \item \textbf{Nodes:} The set of nodes is the union of labeled and unlabeled data, $X_l \cup X_u$.
    \item \textbf{Edges:} Constructed via a $k$-Nearest Neighbor graph (unweighted) 
    or a fully connected graph where weights decay with distance (全连接图,权重随距离衰减):
    $$w_{ij} = \exp\left(-\frac{\|x_i - x_j\|^2}{2\sigma^2}\right)$$
    \item $w_{ij}$ 是高斯核的那个函数，反映了数据点之间的相似度
\end{itemize}

\paragraph{Regularized Classifier}
A standard classifier learns by minimizing a loss term combined with regularization (e.g., LASSO or Regularized Least Squares). We can extend this to semi-supervised learning by using unlabeled data for regularization.

\paragraph{Graph-Based Regularization}
The core principle is that if two instances $x_i$ and $x_j$ are similar (indicated by a large edge weight $w_{ij}$), 
their predicted labels $f(x_i)$ and $f(x_j)$ should also be similar. 
This leads to the following optimization problem: \\
其实就是学习一个学习器来使得损失函数 + 正则化项最小
$$\min_{f} \sum_{i=1}^{l} \ell(y_i, f(x_i)) + \lambda \sum_{i=1}^{N} \sum_{j=1}^{N} w_{ij} \|f(x_i) - f(x_j)\|^2$$

\textbf{第一项：} 有标签数据的损失。
\textbf{第二项：} 图正则化项，强制相似点有相似预测。
\textbf{$\lambda$：} 正则化参数。

% Section 7: Consistency and Topology (image_1b3949.png)
\paragraph{Algorithm Objectives}
The graph-based algorithm operates with two primary goals:
\begin{itemize}
    \item \textbf{Label Constraint:} It enforces the correct labels on the existing labeled data.
    \item \textbf{Manifold Consistency:} It maximizes the consistency of unlabeled examples relative to the underlying graph topology.
\end{itemize}

\subsubsection{Label Propagation Algorithm (标签传播算法)}
\begin{enumerate}
    \item Compute affinity matrix $W$ (亲和矩阵).
    \item Compute degree matrix $D$, where $D_{ii} = \sum_j W_{ij}$.
    \item Initialize $\hat{Y}^{(0)} \leftarrow (y_1, \dots, y_l, 0, 0, \dots, 0)$.
    \item Iterate: $\hat{Y}^{(t+1)} = D^{-1}W\hat{Y}^{(t)}$, fix labels of labeled data.
    \item Repeat until convergence.
    \item Assign labels based on the sign of $\hat{y}_i^{(\infty)}$.
\end{enumerate}

\begin{itemize}
    \item The algorithm forces the labels on the labeled data (强制在标记数据上保留标签)
    \item The algorithm tries to maximizes the consistency of the unlabeled examples with the topology of the graph (试图使未标记示例与图的拓扑结构保持最大程度的一致性)
\end{itemize}

\subsection{Graph Convolutional Networks (GCNs)}

        \item $\mathbf{H}^{(0)} = \mathbf{X} \in \mathbb{R}^{n \times d}$: Input feature matrix.

    \item \textbf{Why GCNs work (optional)}: GCN is an approximated spectral convolution.

            \item Sum: $\mathbf{h}_G = \sum_{i=1}^{n_G} \mathbf{h}_i$; Average: $\mathbf{h}_G = \frac{1}{n_G} \sum_{i=1}^{n_G} \mathbf{h}_i$; Min/Max pooling.
            \item Loss (Supervised): $\mathcal{L} = - \sum_j \sum_{k=1}^K \mathbf{Y}_{jk} \ln \mathbf{Z}_{jk}$.
        \end{itemize}
        \item \textbf{Link Prediction}: Predict new edges in a graph $G$.
        \begin{itemize}
            \item Loss: $\mathcal{L} = - \sum_{(i,j) \in \Omega} (A_{ij} \ln \sigma(\mathbf{z}_i^\top \mathbf{z}_j) + (1 - A_{ij}) \ln(1 - \sigma(\mathbf{z}_i^\top \mathbf{z}_j)))$.
            \item $\sigma$ is the sigmoid function.
        \end{itemize}

\section{Lec6: Non-Linear Dimensionality Reduction}
\subsection{Locally Linear Embedding (LLE)}
\begin{itemize}
    \item \textbf{Core Intuition}: Based on the geometric intuition of \textbf{local linearity}. It assumes that each sample and its neighbors lie on or near a locally linear patch of a low-dimensional manifold(流形) embedded in a high-dimensional space.
    \item \textbf{LLE Assumption}: The projection from high to low dimensionality should preserve neighborhood relationships:
    \begin{itemize}
        \item Projected data points should maintain the same neighbors as in the original space.(人话就是投影后的数据点应该和原空间中的数据点保持相同的邻居关系)
        \item The \textbf{locally linear representation} (reconstruction weights) should be preserved.
    \end{itemize}
    \item \textbf{Algorithm Steps}:
    \begin{enumerate}
        \item \textbf{Find Neighbors}: For each $\mathbf{x}_i$, find its $k$ nearest neighbors $\mathcal{N}_i^k$ ($k \ge d+1$,k 一般比 d 大一点).
        \item \textbf{Compute Weights $\mathbf{W}$}: Solve for weights that reconstruct $\mathbf{x}_i$ from neighbors:
        $$\mathbf{W} = \text{argmin}_{\mathbf{W}} \sum_{i=1}^N \|\mathbf{x}_i - \sum_{j \in \mathcal{N}_i^k} w_{ij} \mathbf{x}_j\|^2, \quad \text{s.t. } \forall i \sum_{j \in \mathcal{N}_i^k} w_{ij} = 1$$
        $w_{i,j}$: locally linear reconstruction weights,局部线性重建权重，且其和为1
        \item \textbf{Compute Projections $\mathbf{Z}$}: Fix $\mathbf{W}$, find low-dimensional $\mathbf{z}_i \in \mathbb{R}^d$:
        $$\mathbf{Z} = \text{argmin}_{\mathbf{Z}} \sum_{i=1}^N \|\mathbf{z}_i - \sum_{j \in \mathcal{N}_i^k} w_{ij} \mathbf{z}_j\|^2, \quad \text{s.t. } \sum \mathbf{z}_i = 0, \frac{1}{N} \mathbf{Z}\mathbf{Z}^\top = \mathbf{I}$$
    \end{enumerate}
    \item \textbf{Solution for $\mathbf{Z}$}: We have:
    $$\sum_{i=1}^{N} \|\mathbf{z}_i - \sum_{j \in \mathcal{N}_i^k} w_{ij}\mathbf{z}_j\|^2 = \|\mathbf{Z} - \mathbf{Z}\mathbf{W}\|_F^2 = \text{trace} \left( \mathbf{Z}(\mathbf{I} - \mathbf{W})(\mathbf{I} - \mathbf{W}^\top)\mathbf{Z}^\top \right)$$
    The cost function is equivalent to $\text{trace}(\mathbf{Z}(\mathbf{I}-\mathbf{W})(\mathbf{I}-\mathbf{W})^\top \mathbf{Z}^\top)$. $\mathbf{Z}$ is composed of the $d$ eigenvectors of $(\mathbf{I}-\mathbf{W})(\mathbf{I}-\mathbf{W})^\top$ corresponding to the \textbf{smallest $d$ non-zero eigenvalues}.\\
    (z 由$(\mathbf{I} - \mathbf{W})(\mathbf{I} - \mathbf{W}^\top)$的最小的d个非零特征值对应的特征向量组成)
    \item \textbf{Notes/Limitations}:
    \begin{itemize}
        \item \textbf{Neighborhood size $k$}: Cannot be too small (insufficient neighbors) or too large (loses local linearity).
        \item \textbf{Out-of-sample extension}: LLE cannot directly handle new data; the entire dataset must be recomputed for new samples.
        (人话就是LLE不能处理新数据, 必须重新计算整个数据集才能处理新样本)
    \end{itemize}
\end{itemize}

\subsection{t-distributed stochastic neighbor embedding(t-SNE)}
Given high-dimensional data $\{\mathbf{x}_1, \dots, \mathbf{x}_N\}$
\begin{itemize}
    \item \textbf{Conditional Probability}: $p_{j|i}$ denotes the probability that $\mathbf{x}_j$ is a neighbor of $\mathbf{x}_i$:
    $$p_{j|i} = \frac{\exp(-\|\mathbf{x}_i - \mathbf{x}_j\|^2 / 2\sigma_i^2)}{\sum_{k \neq i} \exp(-\|\mathbf{x}_i - \mathbf{x}_k\|^2 / 2\sigma_i^2)}, \quad p_{i|i} = 0$$
    \begin{itemize}
        \item $p_{j|i}$: Probability of $\mathbf{x}_j$ being a neighbor of $\mathbf{x}_i$ (i,j 作为邻居的概率)
        \item $\sigma_i$: the size of the neighborhood around $\mathbf{x}_i$ (邻域大小)\\
        (不同的数据点,$、sigma_i$ 不同,其定义了我们试图保留的邻域结构, 根据困惑度 (Perplexity)来计算得出, t-SNE 降维最后的结果与这个值高度相关)
    \end{itemize}
    \item \textbf{Bandwidth $\sigma_i$}: Sets the local neighborhood size, typically determined by user-defined \textbf{Perplexity}.
    \item \textbf{Joint Probability (Symmetry)}: To ensure $p_{ij} = p_{ji}$, define:
    $$p_{ij} = \frac{p_{j|i} + p_{i|j}}{2N}, \quad \text{s.t. } \sum_{i,j} p_{ij} = 1, p_{ii} = 0,(关键性质)$$
    \item \textbf{Perplexity}: A measure of the effective number of neighbors:
    $$\text{perp}(i) = 2^{H(p_{j|i})}, \quad H(p_{j|i}) = -\sum_{j=1}^N p_{j|i} \log p_{j|i}$$
    \begin{itemize}
        \item Low perplexity $\implies$ distribution is good at predicting the sample (focusses on local structure).
        (人话就是低的困惑度意味着分布更好地预测样本, 关注局部结构)
        \item $\sigma_i$ is tuned (e.g., via bisection(二分法)) to match the target perplexity.
        (找到一个$\sigma_i$ ，使得以该点为中心计算出来的 Perplexity 等于用户指定的目标 Perplexity, 二分法是一种有效的方法)
        \item \textbf{Typical values: 5--50}.
    \end{itemize}
    
    \newpage
    \item \textbf{t-SNE: Objective and Optimization}:
    \begin{itemize}
        \item \textbf{Low-dimensional Similarity $q_{ij}$}: Uses Student's t-distribution (d.o.f.=1) to define similarity in embedding space $\mathbf{z}_i \in \mathbb{R}^d$:
        $$q_{ij} = \frac{(1 + \|\mathbf{z}_i - \mathbf{z}_j\|^2)^{-1}}{\sum_{k} \sum_{l \neq k} (1 + \|\mathbf{z}_k - \mathbf{z}_l\|^2)^{-1}}$$
        \item \textbf{Cost Function}: Minimize the KL-divergence between $P$ and $Q$:
        $$\mathcal{L} := \text{KL}(P \| Q) = \sum_{i \neq j} p_{ij} \log \frac{p_{ij}}{q_{ij}}$$
        \item \textbf{Gradient}: Optimization via gradient descent:
        $$\frac{\partial \mathcal{L}}{\partial \mathbf{z}_i} = \sum_{j} (p_{ij} - q_{ij})(\mathbf{z}_i - \mathbf{z}_j)(1 + \|\mathbf{z}_i - \mathbf{z}_j\|^2)^{-1}$$
        \item \textbf{update}: $$\mathbf{z}_i^{(t+1)} = \mathbf{z}_i^{(t)} - \eta \frac{\partial \mathcal{L}}{\partial \mathbf{z}_i}$$
    \end{itemize}
    \item \textbf{Notes/Limitations}:
    \begin{itemize}
        \item \textbf{Visualization}: Primarily used for 2D or 3D visualization.
        \item \textbf{Complexity}: Computationally expensive; suitable for datasets with a few thousand points.
        \item \textbf{Perplexity Sensitivity}: Results depend heavily on perplexity; different values may reveal different structures.
        (结果高度依赖困惑度) 
        \item \textbf{Out-of-sample}: Cannot handle new data directly; requires recomputation of the entire dataset.
    \end{itemize}
\end{itemize}

\subsection{Autoencoder, 自编码器}
\begin{itemize}
    \item \textbf{Core Idea}: A neural network that learns to copy its input to its output. It compresses input into a lower-dimensional latent space (Encoder) and then reconstructs the output (Decoder).
    (人话就是一个神经网络，自我学习将输入复制到输出。它将输入压缩到一个低维的潜在空间（编码器），然后再重构输出（解码器）)
    \item \textbf{Structure}:
    \begin{itemize}
        \item \textbf{Encoder}: $h = f(\mathbf{x}) = \sigma(\mathbf{W}\mathbf{x} + \mathbf{b})$, where $\sigma$ is an activation function (e.g., sigmoid, ReLU).
        \item \textbf{Decoder}: $\hat{\mathbf{x}} = g(h) = \sigma(\mathbf{W}'h + \mathbf{b}')$.
    \end{itemize}
    \item \textbf{Objective}: Minimize reconstruction error(最小化重建误差):
    $$\min_{\theta} \frac{1}{2} \sum_{i=1}^N \|\mathbf{x}_i - g(f(\mathbf{x}_i))\|^2$$
    where $\theta$ represents all weights and biases.
    \item \textbf{Training}: Optimized via backpropagation and gradient descent.
\newpage
    \item \textbf{Variants}:
    \begin{itemize}
        \item \textbf{Stacked AE (SAE), 栈式自编码器}: Multi-layer encoder/decoder for deeper feature learning. Supports \textbf{out-of-sample extension} (unlike PCA/LLE/t-SNE).
        (这个东西可以直接对新数据进行编码)
        \item \textbf{Convolutional AE (CAE), 卷积自编码器}: Replaces fully connected layers with convolutional layers; suitable for image data.
    \end{itemize}
\end{itemize}

\section{Lec7: Generative Models}
\subsection{Introduction}
\begin{itemize}
    \item \textbf{Generative Models:} Learn a probability distribution $p(\mathbf{x})$ from data $\mathcal{D} = \{\mathbf{x}_1, \dots, \mathbf{x}_n\}$ to sample/generate new instances with a similar distribution.
    \item \textbf{Deep Generative Models (DGMs):} Combine generative models with deep neural networks. Achieved SOTA in image/text generation (e.g., ChatGPT).
    \item \textbf{Types of DGMs:}
    \begin{itemize}
        \item \textbf{AutoEncoder (AE)}\\
        \textbf{Recap}: \\ 
        An unsupervised network for identity transformation. 
        It maps high-dimensional $\mathbf{x}$ to latent $\mathbf{z}$ 
        via encoder $f_\phi$, then reconstructs $\mathbf{x}'$ via decoder $g_\theta$. \\
        Targeted loss:
         $\min_{\phi,\theta} \frac{1}{n} \sum_{i=1}^n \|\mathbf{x}_i - g_\theta(f_\phi(\mathbf{x}_i))\|^2 + \mathcal{R}(\phi, \theta)$.
        \item Variational AutoEncoder (VAE)
        \item Generative Adversarial Networks (GANs)
        \item Normalizing Flows
        \item Diffusion Models
    \end{itemize}
\end{itemize}

\subsection{Variational AutoEncoder (VAE)}
\begin{itemize}
    \item \textbf{Concept:} VAE transforms input $\mathbf{x}$ into a prior distribution $p_z$ 
    using encoder $f_\phi$, then reconstructs it via decoder $g_\theta$.
    (人话就是 VAE 输入了一个分布，将他映射到另一个分布上，不像 AE 直接映射到一个点上)
    \item \textbf{Generation:} Given trained $f_\phi$ and $g_\theta$, a new sample is generated by:
    \begin{enumerate}
        \item Sample $\mathbf{z}_i \sim p_z$.
        \item Generate $\hat{\mathbf{x}}_i = g_\theta(\mathbf{z}_i)$.
    \end{enumerate}
    \item \textbf{Objective:} Maximize the log-likelihood of real data samples $\mathbf{X} = \{\mathbf{x}_1, \dots, \mathbf{x}_n\}$:
    $$\theta^* = \arg \max_\theta \sum_{i=1}^n \log p_\theta(\mathbf{x}_i)$$
    where $p_\theta(\mathbf{x}_i)$ denotes the probability of generating $\mathbf{x}_i$ given parameters $\theta$.

    \item \textbf{Marginal Likelihood:} Direct computation is intractable because integrating over all $\mathbf{z}$ is impossible:
    $$p_\theta(\mathbf{x}) = \int p_\theta(\mathbf{x}, \mathbf{z})d\mathbf{z} = \int p_\theta(\mathbf{x}|\mathbf{z})p_\theta(\mathbf{z})d\mathbf{z}$$
    Using Bayes' theorem, $p_\theta(\mathbf{x}) = \frac{p_\theta(\mathbf{x}|\mathbf{z})p_\theta(\mathbf{z})}{p_\theta(\mathbf{z}|\mathbf{x})}$, but the posterior $p_\theta(\mathbf{z}|\mathbf{x})$ cannot be computed.
    
    \item \textbf{Variational Inference:} Train another neural network (encoder) $f_\phi$ to learn $q_\phi(\mathbf{z}|\mathbf{x}) \approx p_\theta(\mathbf{z}|\mathbf{x})$.
    
    \item \textbf{Log-Likelihood Decomposition:} Introducing $q_\phi(\mathbf{z}|\mathbf{x})$, we decompose $\log p_\theta(\mathbf{x})$:
    \begin{align*}
        \log p_\theta(\mathbf{x}) &= \log \frac{p_\theta(\mathbf{x}|\mathbf{z})p(\mathbf{z})}{p_\theta(\mathbf{z}|\mathbf{x})} = \log \frac{p_\theta(\mathbf{x}|\mathbf{z})p(\mathbf{z})q_\phi(\mathbf{z}|\mathbf{x})}{p_\theta(\mathbf{z}|\mathbf{x})q_\phi(\mathbf{z}|\mathbf{x})} \\
        &= \log p_\theta(\mathbf{x}|\mathbf{z}) - \log \frac{q_\phi(\mathbf{z}|\mathbf{x})}{p(\mathbf{z})} + \log \frac{q_\phi(\mathbf{z}|\mathbf{x})}{p_\theta(\mathbf{z}|\mathbf{x})}
    \end{align*}
    Taking the expectation $\mathbb{E}_{\mathbf{z} \sim q_\phi(\mathbf{z}|\mathbf{x})}$:
    \begin{align*}
        \log p_\theta(\mathbf{x}) &= \mathbb{E}_{\mathbf{z} \sim q_\phi(\mathbf{z}|\mathbf{x})}[\log p_\theta(\mathbf{x})] = \int q_\phi(\mathbf{z}|\mathbf{x}) \log p_\theta(\mathbf{x}) d\mathbf{z} \\
        &= \int q_\phi(\mathbf{z}|\mathbf{x}) \log p_\theta(\mathbf{x}|\mathbf{z}) d\mathbf{z} - \int q_\phi(\mathbf{z}|\mathbf{x}) \log \frac{q_\phi(\mathbf{z}|\mathbf{x})}{p(\mathbf{z})} d\mathbf{z} + \int q_\phi(\mathbf{z}|\mathbf{x}) \log \frac{q_\phi(\mathbf{z}|\mathbf{x})}{p_\theta(\mathbf{z}|\mathbf{x})} d\mathbf{z} \\
        &= \mathbb{E}_{\mathbf{z} \sim q_\phi(\mathbf{z}|\mathbf{x})}[\log p_\theta(\mathbf{x}|\mathbf{z})] - D_{KL}(q_\phi(\mathbf{z}|\mathbf{x}) \| p(\mathbf{z})) + D_{KL}(q_\phi(\mathbf{z}|\mathbf{x}) \| p_\theta(\mathbf{z}|\mathbf{x}))
    \end{align*}

    \item \textbf{Evidence Lower Bound (ELBO):} Since the KL-divergence is always non-negative ($D_{KL} \ge 0$), we have:
    $$\log p_\theta(\mathbf{x}) \ge \mathcal{L}_{\phi, \theta}(\mathbf{x}) = \underbrace{\mathbb{E}_{\mathbf{z} \sim q_\phi(\mathbf{z}|\mathbf{x})}[\log p_\theta(\mathbf{x}|\mathbf{z})]}_{\text{Reconstruction Loss}} - \underbrace{D_{KL}(q_\phi(\mathbf{z}|\mathbf{x}) \| p(\mathbf{z}))}_{\text{KL Divergence}}$$
    where $\mathcal{L}_{\phi, \theta}(\mathbf{x})$ is the Evidence Lower Bound (or variational lower bound). It follows that:
    $$\log p_\theta(\mathbf{x}) = \mathcal{L}_{\phi, \theta}(\mathbf{x}) + D_{KL}(q_\phi(\mathbf{z}|\mathbf{x}) \| p_\theta(\mathbf{z}|\mathbf{x}))$$
    
    \item \textbf{VAE Objective:} VAE maximizes the ELBO:
    $$\phi^*, \theta^* = \arg \max_{\phi, \theta} \mathbb{E}_{\mathbf{z} \sim q_\phi(\mathbf{z}|\mathbf{x})}[\log p_\theta(\mathbf{x}|\mathbf{z})] - D_{KL}(q_\phi(\mathbf{z}|\mathbf{x}) \| p(\mathbf{z}))$$
    When $\mathcal{L}_{\phi, \theta}(\mathbf{x}) = \log p_\theta(\mathbf{x})$, we have $D_{KL}(q_\phi(\mathbf{z}|\mathbf{x}) \| p_\theta(\mathbf{z}|\mathbf{x})) = 0$, meaning $q_\phi(\mathbf{z}|\mathbf{x}) = p_\theta(\mathbf{z}|\mathbf{x})$.

    (这一堆用人话说就是：由于那个 $log p_\theta(x)$ 是无法计算的，我们引入了一个新的分布 $q_\phi(z|x)$，
    来近似那个无法计算的后验分布 $p_\theta(z|x)$。这样得到了一个下界（ELBO），
    我们最大化这个下界来训练我们的模型。
    当这个下界等于 $log p_\theta(x)$ 的时候，我们就达到了最优，
    此时 $q_\phi(z|x)$ 就等于 $p_\theta(z|x)$，也就是说我们成功地近似了那个无法计算的后验分布)(并非人话，还是看不懂)\\
    (牛魔的这都是什么东西。)

    \item \textbf{Reparameterization Trick}:
    \begin{itemize}
        \item \textbf{Problem}: During training, we need to sample $\mathbf{z}$ 
        from $q_\phi(\mathbf{z}|\mathbf{x}) = \mathcal{N}(\mu, \sigma^2)$ 
        to compute the reconstruction loss. 
        However, the sampling operation is non-differentiable (\textbf{a stochastic process}), 
        preventing gradients from flowing back to $\mu$ and $\sigma$, 
        which makes backpropagation impossible.
    \end{itemize}

    \begin{figure}[H]
        \centering
        \includegraphics[width=0.8\textwidth]{figures/VAE optimization.png}
        \caption{VAE Optimization and ELBO}
    \end{figure}

    \item \textbf{Reparameterization Trick(Cont.)}:
    \begin{itemize}
        \item \textbf{Solution:} Sample random noise $\epsilon \sim \mathcal{N}(0, \mathbf{I})$ 
        and compute $\mathbf{z} = \mu + \sigma \odot \epsilon$.
        \item \textbf{Proof of Mean:} By linearity of expectation $E[aX + b] = aE[X] + b$:
        \begin{align*}
            E[\mathbf{z}] &= E[\mu + \sigma \cdot \epsilon] = \mu + \sigma \cdot E[\epsilon] \\
            &= \mu + \sigma \cdot 0 = \mu
        \end{align*}
        \item \textbf{Proof of Variance:} By variance property $Var(aX + b) = a^2 Var(X)$:
        \begin{align*}
            Var(\mathbf{z}) &= Var(\mu + \sigma \cdot \epsilon) = \sigma^2 \cdot Var(\epsilon) \\
            &= \sigma^2 \cdot 1 = \sigma^2
        \end{align*}
        \item \textbf{Result:} Randomness is shifted to $\epsilon$.
        Since $\mu$ and $\sigma$ are deterministic and differentiable transformations, 
        gradients can flow from $\mathbf{z}$ to $\mu$ and $\sigma$, then back to the encoder.
    \end{itemize}

    \item \textbf{The Details of the Loss Function}:
    \begin{itemize}
        \item \textbf{KL Divergence Term:} Considering one element of $\mathbf{z}$:
        \begin{align*}
            D_{KL} (q_\phi (z|x) \| p(z)) &= \int \frac{1}{\sqrt{2\pi\sigma_q^2}} 
            \exp \left( -\frac{(z - \mu_q)^2}{2\sigma_q^2} \right) \log \left( \frac{\frac{1}{\sqrt{2\pi\sigma_p^2}} \exp \left( -\frac{(z - \mu_p)^2}{2\sigma_p^2} \right)}
            {\frac{1}{\sqrt{2\pi\sigma_q^2}} \exp \left( -\frac{(z - \mu_q)^2}{2\sigma_q^2} \right)} \right) dz \\
            &= \log \left( \frac{\sigma_q}{\sigma_p} \right) - \frac{\sigma_q^2 + (\mu_q - \mu_p)^2}{2\sigma_p^2} + \frac{1}{2} \\
            &= \frac{1}{2} \left[ 1 + \log(\sigma_q^2) - \sigma_q^2 - \mu_q^2 \right] \quad \text{(when } p(z) \sim \mathcal{N}(0, 1))
        \end{align*}
        \item \textbf{Reconstruction Term:} Assuming $p_\theta(\mathbf{x}|\mathbf{z}) = \mathcal{N}(\mathbf{x}; g_\theta(\mathbf{z}), \Sigma_{\mathbf{x}})$:
        $$\mathbb{E}_{\mathbf{z} \sim q_\phi(\mathbf{z}|\mathbf{x})}[\log p_\theta(\mathbf{x}|\mathbf{z})] \propto -\|\mathbf{x} - g_\theta(\mathbf{z})\|^2$$
    \end{itemize} 

    (人话就是，KL散度最后变成了那个非常简单的形式。重构损失等价于输入和输出之间的均方误差，前面那一大堆积分消失了。) \\
    (虽然我也不知道这牛魔怎么推出来的)
\end{itemize}

\subsection{Generative Adversarial Networks (GANs)}
\begin{itemize}
    \item Inspired by game theory, GAN involves an adversarial process training two networks:
    \begin{itemize}
        \item \textbf{Generator $G$}: Learns the real data distribution to produce realistic samples.
        \item \textbf{Discriminator $D$}: Estimates the probability that a sample is real rather than from $G$.
    \end{itemize}
    \item \textbf{Objective}: $D$ aims to maximize correct labeling of both real and generated samples,
    while $G$ aims to maximize $D$'s mistake probability. \\
    (人话就是生成器和判别器在比谁牛逼，生成器想骗过判别器，判别器想识破生成器)

    \item \textbf{Training Steps}:
    \begin{enumerate}
        \item Fix $G$, train $D$ to better distinguish samples.
        \item Fix $D$, train $G$ to produce more realistic samples.
        \item Iterate steps 1 and 2 until convergence.\\
        (其实还是回合制游戏)
    \end{enumerate}
    \item \textbf{Minimax Game}:
    \begin{itemize}
        \item \textbf{Notations}: $p_r$ (real distribution), $p_g$ (generator distribution), $p_z$ (noise prior).
        \item \textbf{Discriminator Objective}: Maximize $D(x)$ for $x \sim p_r$ and minimize $D(G(z))$ for $z \sim p_z$:
        $$\max_D \mathbb{E}_{\mathbf{x} \sim p_r(\mathbf{x})}[\log D(\mathbf{x})] + \mathbb{E}_{\mathbf{z} \sim p_z(\mathbf{z})}[\log(1 - D(G(\mathbf{z})))]$$
        (判定真实数据准确并且判断生成器生成的数据不准确，提高这个概率)
        \item \textbf{Generator Objective}: Minimize the probability of $D$ correctly identifying fake samples:
        $$\min_G \mathbb{E}_{\mathbf{z} \sim p_z(\mathbf{z})}[\log(1 - D(G(\mathbf{z})))]$$ \\
        (生成器想让判别器判断生成的数据为真，提高这个概率)
        \item \textbf{Combined Loss Function} $\mathcal{L}(G, D)$:
        $$\min_G \max_D \mathcal{V}(D, G) = \mathbb{E}_{\mathbf{x} \sim p_r(\mathbf{x})}[\log D(\mathbf{x})] + \mathbb{E}_{\mathbf{z} \sim p_z(\mathbf{z})}[\log(1 - D(G(\mathbf{z})))]$$\\
        (来一场痛快的回合制游戏口牙)
    \end{itemize}

    \item \textbf{Pros \& Cons}:
    \begin{itemize}
        \item \textbf{Advantages}: Direct sampling; avoids MLE; yields more realistic samples than VAE.
        \item \textbf{Disadvantages}: Implicit distribution (cannot compute $p(\mathbf{x})$); hard to train (no convergence guarantee, prone to mode collapse).
    \end{itemize}
\end{itemize}

\subsection{Diffusion Models}
\begin{itemize}
    \item \textbf{Forward Diffusion Process,前向扩散}:
    \begin{itemize}
        \item Given data $\mathbf{x}_0 \sim q(\mathbf{x})$, iteratively add Gaussian noise in $T$ steps:
        $$\mathbf{x}_t = \sqrt{1 - \beta_t}\mathbf{x}_{t-1} + \sqrt{\beta_t}\epsilon_{t-1}$$
        where $\{\beta_t \in (0,1)\}_{t=1}^T$ is a variance schedule and $\epsilon_t \sim \mathcal{N}(\mathbf{0}, \mathbf{I})$.
        \item Transition probability:
        $$q(\mathbf{x}_t|\mathbf{x}_{t-1}) = \mathcal{N}(\mathbf{x}_t; \sqrt{1 - \beta_t}\mathbf{x}_{t-1}, \beta_t\mathbf{I})$$
        \item As $T \to \infty$, $\mathbf{x}_T$ becomes an isotropic Gaussian.
    \end{itemize}
    
    \item \textbf{Closed-Form Sampling at Arbitrary Substep $t$}:
    \begin{itemize}
        \item Let $\alpha_t = 1 - \beta_t$ and $\bar{\alpha}_t = \prod_{s=1}^t \alpha_s$.
        \item Recursive derivation:
        \begin{align*}
            \mathbf{x}_t &= \sqrt{\alpha_t}\mathbf{x}_{t-1} + \sqrt{1 - \alpha_t}\epsilon_{t-1} \\
            &= \sqrt{\alpha_t\alpha_{t-1}}\mathbf{x}_{t-2} + \sqrt{1 - \alpha_t\alpha_{t-1}}\bar{\epsilon}_{t-2} \\
            &= \cdots \\
            &= \sqrt{\bar{\alpha}_t}\mathbf{x}_0 + \sqrt{1 - \bar{\alpha}_t}\epsilon
        \end{align*}
        where $\epsilon \sim \mathcal{N}(\mathbf{0}, \mathbf{I})$.
        \item Conditional distribution of $\mathbf{x}_t$ given $\mathbf{x}_0$:
        $$q(\mathbf{x}_t|\mathbf{x}_0) = \mathcal{N}(\mathbf{x}_t; \sqrt{\bar{\alpha}_t}\mathbf{x}_0, (1 - \bar{\alpha}_t)\mathbf{I})$$
    (人话： $x_t$ 就是 $x_0$ 和随机噪声 $\epsilon$ 的一个加权平均)
    \end{itemize}

    \item \textbf{Reverse Diffusion Process，反向去噪}:
    \begin{itemize}
        \item The true reverse process $q(\mathbf{x}_{t-1}|\mathbf{x}_t)$ is unknown. If $\beta_t$ is small enough, it is also Gaussian.
        \item We train a neural network $p_\theta$ to approximate it:
        $$p_\theta(\mathbf{x}_{t-1}|\mathbf{x}_t) = \mathcal{N}(\mathbf{x}_{t-1}; \mu_\theta(\mathbf{x}_t, t), \Sigma_\theta(\mathbf{x}_t, t))$$
        \item \textbf{Tractable Reverse Conditional Probability}: When conditioned on $\mathbf{x}_0$, the reverse probability has a closed form:
        $$q(\mathbf{x}_{t-1}|\mathbf{x}_t, \mathbf{x}_0) = \mathcal{N}(\mathbf{x}_{t-1}; \tilde{\mu}_t(\mathbf{x}_t, \mathbf{x}_0), \tilde{\beta}_t\mathbf{I})$$
        where the parameters are:
        $$\tilde{\beta}_t = \frac{1 - \bar{\alpha}_{t-1}}{1 - \bar{\alpha}_t}\beta_t, \quad \tilde{\mu}_t(\mathbf{x}_t, \mathbf{x}_0) = \frac{\sqrt{\alpha_t}(1 - \bar{\alpha}_{t-1})}{1 - \bar{\alpha}_t}\mathbf{x}_t + \frac{\sqrt{\bar{\alpha}_{t-1}}\beta_t}{1 - \bar{\alpha}_t}\mathbf{x}_0$$
    \end{itemize}

    \newpage
    \item \textbf{Training Objective(optional)}:
    \begin{itemize}
        \item Minimize the variational bound on negative log-likelihood:
        \begin{align*}
        \mathbb{E}[-\log p_\theta(\mathbf{x}_0)] 
        &\le \mathbb{E}_q \left[ -\log \frac{p_\theta(\mathbf{x}_{0:T})}{q(\mathbf{x}_{1:T}|\mathbf{x}_0)} \right] \\
        &= \mathbb{E}_q \left[ -\log p(\mathbf{x}_T) - \sum_{t\ge 1} \log \frac{p_\theta(\mathbf{x}_{t-1}|\mathbf{x}_t)}{q(\mathbf{x}_t|\mathbf{x}_{t-1})} \right] \\
        &= \mathbb{E}_q \Big[ \underbrace{D_{KL}(q(\mathbf{x}_T|\mathbf{x}_0) \| p(\mathbf{x}_T))}_{L_T} \\
        &\quad + \sum_{t>1} \underbrace{D_{KL}(q(\mathbf{x}_{t-1}|\mathbf{x}_t, \mathbf{x}_0) \| p_\theta(\mathbf{x}_{t-1}|\mathbf{x}_t))}_{L_{t-1}} - \underbrace{\log p_\theta(\mathbf{x}_0|\mathbf{x}_1)}_{L_0} \Big]
        \end{align*}
        \item For $L_{t-1}$, using reparameterization:
        \begin{align*}
        L_{t-1} 
        &= \mathbb{E}_q \left[ \frac{1}{2\sigma_t^2} \| \tilde{\mu}_t(\mathbf{x}_t, \mathbf{x}_0) - \mu_\theta(\mathbf{x}_t, t) \|^2 \right] + C \\
        &= \mathbb{E}_{\mathbf{x}_0, \epsilon} \left[ \frac{1}{2\sigma_t^2} \left\| \frac{1}{\sqrt{\alpha_t}} \left( \mathbf{x}_t(\mathbf{x}_0, \epsilon) - \frac{\beta_t}{\sqrt{1-\bar{\alpha}_t}}\epsilon \right) - \mu_\theta(\mathbf{x}_t(\mathbf{x}_0, \epsilon), t) \right\|^2 \right] + C \\
        &= \mathbb{E}_{\mathbf{x}_0, \epsilon} \left[ \frac{\beta_t^2}{2\sigma_t^2\alpha_t(1-\bar{\alpha}_t)} \left\| \epsilon - \epsilon_\theta \left( \sqrt{\bar{\alpha}_t}\mathbf{x}_0 + \sqrt{1-\bar{\alpha}_t}\epsilon, t \right) \right\|^2 \right] + C
        \end{align*}
        \item \textbf{Simplified Objective} (Optimization via SGD):
        $$L_{\text{simple}}(\theta) := \mathbb{E}_{t, \mathbf{x}_0, \epsilon} \left[ \left\| \epsilon - \epsilon_\theta \left( \sqrt{\bar{\alpha}_t}\mathbf{x}_0 + \sqrt{1-\bar{\alpha}_t}\epsilon, t \right) \right\|^2 \right]$$
    \end{itemize}
    (反正是 optional, 摆了, 当个吉祥物)

    \item \textbf{Advantages \& Disadvantages}:
    \begin{itemize}
        \item \textbf{Pros}: Higher generation quality than VAE/GAN; Explicit probability distribution.
        \item \textbf{Cons}: Time-consuming training; Slow sampling generation due to large $T$ (e.g., $T=1000$).
    \end{itemize}

\end{itemize}

\section{Lec8: Foundation Models}
\subsection{Introduction}
\begin{itemize}
    \item \textbf{Definition}: Large-scale pre-trained models serving as a general-purpose basis for downstream tasks.
    \item \textbf{Adaptation}: Trained on vast data, fine-tuned for specific applications via transfer learning, prompting, or few-shot learning.
    \item \textbf{Applications}: Key to modern NLP (e.g., BERT, GPT-3, CLIP) and CV (e.g., DALL$\cdot$E, Stable Diffusion, BEiT, SORA).
    \item \textbf{Architecture}: Built on multiple neural networks like MLP, CNN, RNN, and Transformer.
\end{itemize}

\subsection{BERT (2018)}
\begin{itemize}
    \item \textbf{Concept}: Bidirectional Encoder Representations from Transformers, designed for context understanding.
    \item \textbf{Architecture}: Encoder-only Transformer.
    \item \textbf{Pre-training}: Masked Language Modeling (MLM) and Next Sentence Prediction (NSP).
    \item \textbf{Data/Training}: Trained on BookCorpus and Wikipedia; Base (110M) and Large (340M) versions.
    \item \textbf{Fine-tuning}: Adapt to specific tasks (e.g., Sentiment Analysis, QA) 
    with fewer resources and smaller datasets.
    \item 一句话就是，BERT 只有 Transformer 的 Encoder 部分，专注于根据上下文理解文本。
    核心用处是可以判断某个词在当前语境的意思，或者句子之间的关系。

\end{itemize}
\subsection{GPT-3 (2020)}
\begin{itemize}
    \item \textbf{Concept}: Large language model by OpenAI, capable of human-like text generation and diverse NLP tasks.
    \item \textbf{Architecture}: Decoder-only Transformer; 175B parameters.
    \item \textbf{Pre-training}: Self-supervised learning via Next Word Prediction (causal language modeling).
    \item \textbf{Data}: Trained on hundreds of billions of words (Common Crawl, WebText2, Books, Wikipedia).
    \item \textbf{Capabilities}: Strong zero/few-shot performance in coding (CSS, Python, etc.), translation, and summarization.
    \item 一句话就是， GPT-3 只有 Transformer 的 Decoder 部分，专注于根据前文生成文本。
    训练方式是预测下一个词是什么，可以用来续写文本，或者根据提示生成文本。
\end{itemize}

\subsection{CLIP (2021)}
\begin{itemize}
    \item \textbf{Concept}: Contrastive Language-Image Pre-training by OpenAI.
    \item \textbf{Objective}: Align text and image representations in a shared vector space, where similar pairs are close and dissimilar ones are far apart.
    \item \textbf{Architecture}: Consists of a \textbf{Visual Encoder} (ViT or ResNet) and a \textbf{Text Encoder} (Transformer).
    \item \textbf{Training}: Contrastive pre-training on "WebImageText" (WIT) dataset with 400M image-caption pairs.
    \item \textbf{Loss Function}: Optimized via a symmetric cross-entropy loss over the similarity matrix:
    \[ L = -\frac{1}{N} \sum_{i} \ln \frac{e^{v_i^T w_i / T}}{\sum_{j} e^{v_i^T w_j / T}} - \frac{1}{N} \sum_{j} \ln \frac{e^{v_j^T w_j / T}}{\sum_{i} e^{v_i^T w_j / T}} \]
    \item \textbf{Applications}: Image captioning, text-to-image generation, and zero-shot prediction (using label text as prompts).
    \item 一句话：通过对比学习将文本和图像映射到同一特征空间。
    训练时让匹配的图文对向量尽可能接近，不匹配的远离。
\end{itemize}
\subsection{Stable Diffusion (2022)}
\begin{itemize}
    \item \textbf{Concept}: A deep learning, text-to-image model by Stability AI based on Latent Diffusion Models (LDM).
    \item \textbf{Architecture}: VAE + UNet + CLIP + Diffusion.
    \item \textbf{Process}:
    \begin{itemize}
        \item \textbf{Pixel Space}: Image $x$ is encoded into latent space $z$ by VAE Encoder $\mathcal{E}$, and decoded back to pixel space $\tilde{x}$ by Decoder $\mathcal{D}$.
        \item \textbf{Latent Space}: The diffusion process (adding noise) and denoising process occur here. 
        \item \textbf{Denoising U-Net $\epsilon_\theta$}: Predicts and removes noise using cross-attention mechanisms ($Q, K, V$) and skip connections.
        \item \textbf{Conditioning}: External guidance (Text, Semantic Map, Images) is injected via a conditioning encoder $\tau_\theta$ into the U-Net.
    \end{itemize}
    \item \textbf{Scale}: Version XL contains 3.5 billion parameters.
    \item 一句话：在潜空间 (Latent Space) 进行扩散和去噪的文生图模型，利用 VAE 降维以提高效率，并通过 Cross-Attention 引入文本引导。
\end{itemize}
\subsection{LoRA: Low-Rank Adaptation}
\begin{itemize}
    \item \textbf{Concept}: Efficiently fine-tune large pre-trained models by decomposing the weight update matrix into two low-rank matrices.
    \item \textbf{Mathematical Formulation}:
    \begin{itemize}
        \item For a pre-trained weight matrix $W \in \mathbb{R}^{d \times d}$, the output is modified as: $h = Wx + \Delta Wx = Wx + ABx$
        \item where $A \in \mathbb{R}^{d \times r}$ and $B \in \mathbb{R}^{r \times d}$ are trainable parameters, with rank $r \ll d$.
        \item Full-parameter update loss: $\mathcal{L} = \frac{1}{2} \| Y - WX \|_F^2$, gradient complexity $\mathcal{O}(d^3)$.
        \item LoRA loss: $\mathcal{L} = \frac{1}{2} \| Y - (W + AB)X \|_F^2$.
    \end{itemize}
    \item \textbf{Optimization}:
    \begin{itemize}
        \item Gradients: $\frac{\partial \mathcal{L}}{\partial A} = -(Y - WX - ABX)(BX)^T$ and $\frac{\partial \mathcal{L}}{\partial B} = -A^T (Y - WX - ABX)X^T$.
        \item Complexity: Total time complexity reduced to $\mathcal{O}(d^2 r + r^2 d)$, which is significantly less than $\mathcal{O}(d^3)$ when $r \ll d$.
    \end{itemize}
    \item 一句话：不改变预训练权重 $W$，只训练降维后的 $A$ 和 $B$。极大地减少了计算开销和显存占用。
\end{itemize}

\subsection{Other Fine-tuning Strategies}
\begin{itemize}
    \item \textbf{Partial Fine-tuning}: Fine-tune only a subset of parameters (e.g., top layers or specific layers) while freezing the rest.
    \item \textbf{Multi-task Fine-tuning}: Train the model on multiple tasks simultaneously to improve generalization.
    \item \textbf{Adversarial Fine-tuning}: Incorporate adversarial examples to enhance model robustness.
\end{itemize}

\subsection{Model Compression}
\begin{itemize}
    \item \textbf{Knowledge Distillation，知识蒸馏}: A pre-trained Teacher model guides the training of a smaller Student model, transferring generalization ability at a reduced scale.
    \begin{itemize}
        \item \textbf{Loss function}: $\mathcal{L} = \mathcal{L}_{\text{CE}} + \lambda \cdot T^2 \cdot \mathcal{L}_{\text{KL}}$
        \item \textbf{Student loss (CE)}: $\mathcal{L}_{\text{CE}} = -\sum_i y_i \log(q_i)$, where $y$ is the ground-truth label and $q$ is the Student's predicted probability, $q_i = \frac{\exp(z_i / T)}{\sum_j \exp(z_j / T)}$.
        \item \textbf{Distillation loss (KL)}: $\mathcal{L}_{\text{KL}} = \sum_i p_i \log\left(\frac{p_i}{q_i}\right)$, where $p$ is the Teacher's predicted probability (soft target at temperature $T=t$).
        \item \textbf{Temperature $T$}: A higher $T$ smooths the softmax output, letting the Student learn inter-class similarity (dark knowledge). Use $T=1$ at inference.
        \item \textbf{Pros}: Significantly reduces model size while retaining the Teacher's generalization ability.
        \item \textbf{Cons}: Relies on a high-quality Teacher and requires additional training time.
    \end{itemize}
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.9\linewidth]{Knowledge Distillation.png}
    \end{figure}

    \item \textbf{Quantization，量化}: Convert model weights and activations from floating-point to lower-precision formats to reduce memory and accelerate inference.
    \begin{itemize}
        \item \textbf{Formats} (bits = sign + exponent + mantissa):
        \begin{itemize}
            \item \texttt{fp32}: 1 + 8 + 23 bits
            \item \texttt{fp16}: 1 + 5 + 10 bits
            \item \texttt{bfloat16}: 1 + 8 + 7 bits (same exponent range as fp32, reduced precision)
            \item \texttt{int8}: 8 bits integer; \texttt{int4}: 4 bits integer
        \end{itemize}
        \item \textbf{Example} (3.12345): \texttt{fp16} $\to$ \texttt{0 10000 1000111110}; \texttt{int4} $\to$ \texttt{0011}
        \item \textbf{Pros}: Greatly reduces memory usage and accelerates inference.
        \item \textbf{Cons}: Aggressive quantization may lead to significant accuracy loss.
    \end{itemize}

    \item \textbf{Factorization，分解}: Decompose large weight matrices into smaller ones.
    \begin{itemize}
        \item \textbf{Pros}: Mathematically sound; significantly reduces parameters.
        \item \textbf{Cons}: Decomposition may introduce errors, requiring retraining to recover performance.
    \end{itemize}

    \item \textbf{Pruning，剪枝}: Remove redundant parameters or structures to simplify the network.
    \begin{itemize}
        \item \textbf{Pros}: Significantly reduces parameters and computation; may improve generalization.
        \item \textbf{Cons}: Unstructured pruning requires specialized hardware to support sparse computation.
    \end{itemize}
\end{itemize}

\subsection{Challenges}
\begin{itemize}
    \item \textbf{Introduction}
    \begin{itemize}
        \item \textbf{Hallucination}: LLMs occasionally generate plausible but incorrect content that (1) diverges from user's input, (2) misaligns with established knowledge, or (3) contradicts previously generated context.
        \item \textbf{Interpretability}: Difficult to understand why a model produces a given output.
        \item \textbf{Limited context length}: Cannot process arbitrarily long inputs.
        \item \textbf{High compute requirements}: Training and inference demand significant hardware resources.
    \end{itemize}

    \item \textbf{Hallucination Detection}
    \begin{itemize}
        \item \textbf{Perplexity (information theory)}: Uncertainty measure for a discrete distribution:
        $\mathrm{PPL}(p) = b^{-\sum_x p(x)\log_b p(x)}$, where $b \in \{2, 10, e, \ldots\}$.
        \item \textbf{Perplexity (LLM)}: For a token sequence $\mathbf{x}=(x_1,\ldots,x_n)$, an LLM with parameters $\theta$ models $p_\theta(x_i \mid \mathbf{x}_{<i})$. Perplexity is:
        \[
            \mathrm{PPL}_\theta(\mathbf{x}) = \exp\!\left(-\frac{1}{n}\sum_{i=1}^{n}\log P_\theta(x_i \mid \mathbf{x}_{<i})\right)
        \]
        It quantifies the model's uncertainty when encountering new tokens.
        \item \textbf{Applications of $\mathrm{PPL}_\theta$}:
        \begin{itemize}
            \item \textbf{Training signal}: Minimizing PPL $\equiv$ minimizing cross-entropy loss.
            \item \textbf{Benchmarking}: Compare models on standard datasets (e.g., WikiText-2); lower PPL is better.
            \item \textbf{Domain check}: Identify which base model best fits a target domain before fine-tuning.
            \item \textbf{Fluency filter}: Unnatural sentences yield high PPL.
        \end{itemize}
        \item \textbf{Limitations of PPL}: Measures statistical fluency, not factual accuracy — a model can achieve low PPL while generating false statements.
        \item \textbf{EigenScore} {[Chen et al., 2024]}: $E = \frac{1}{K}\sum_{i}^{K} \log(\lambda_i)$, where $\lambda_i$ are eigenvalues; captures semantic consistency across multiple generated responses.
    \end{itemize}

\end{itemize}

\section{Lec9: AI for Science}
\subsection{AI for Physics}
\begin{itemize}
    \item \textbf{Physical Laws and PDEs}: Most fundamental physical laws are expressed as PDEs. General form:
    \[
        F\!\left(x,y,\ldots,t,\,u,\,\frac{\partial u}{\partial x},\frac{\partial u}{\partial y},\ldots,\frac{\partial^2 u}{\partial x^2},\frac{\partial^2 u}{\partial y^2},\frac{\partial u}{\partial t},\frac{\partial^2 u}{\partial t^2},\ldots\right)=0
    \]
    ODEs are special, simplified cases. Examples:
    \begin{itemize}
        \item \textbf{Newton's Second Law}: $F = m\dfrac{d^2x}{dt^2}$
        \item \textbf{Heat Transfer}: $\dfrac{\partial u}{\partial t} = \alpha \nabla^2 u$, where $u(x,y,z,t)$ is temperature, $\nabla^2 u = \dfrac{\partial^2 u}{\partial x^2}+\dfrac{\partial^2 u}{\partial y^2}+\dfrac{\partial^2 u}{\partial z^2}$ (Laplacian, spatial curvature), $\alpha$ is thermal diffusivity.
        \item \textbf{Schr\"{o}dinger Equation}: $i\hbar\dfrac{\partial\psi}{\partial t} = -\dfrac{\hbar^2}{2m}\nabla^2\psi + V\psi$, where $\psi(x,y,z,t)$ is the wave function (probability amplitude), $i\hbar\frac{\partial\psi}{\partial t}$ is time evolution, $-\frac{\hbar^2}{2m}\nabla^2\psi$ is kinetic energy, $V\psi$ is potential energy.
        \item \textbf{Fluid Mechanics}: Navier-Stokes Equations.
        \item \textbf{Electromagnetism}: Maxwell's Equations.
    \end{itemize}

    \item \textbf{Solving PDEs}:
    \begin{itemize}
        \item \textbf{Simple PDEs} (e.g., Heat Conduction): analytical solutions, e.g., separation of variables.
        \item \textbf{Classical methods} (complex PDEs): FEM, FVM, FDM. Limitation: require manual mesh creation; computationally intractable for high-dimensional problems.
        \item \textbf{Modern AI-based methods}: Neural Operators, Physics-Informed Neural Networks (PINNs). Advantages: mesh-free, overcome the curse of dimensionality, integrate physical laws with sparse data for inverse problems.
    \end{itemize}

    \item \textbf{Physics-Informed Neural Networks (PINNs)}: Universal function approximators that embed physical laws (PDEs) into the learning process. General PDE form:
    \[
        \mathcal{F}(u(z);\gamma) = f(z) \quad z \in \Omega, \qquad \mathcal{B}(u(z)) = g(z) \quad z \in \partial\Omega
    \]
    \begin{itemize}
        \item[$\Omega$:] domain $\subset \mathbb{R}^d$ with boundary $\partial\Omega$
        \item[$z$:] $:= [x_1,\ldots,x_{d-1};t]$, space-time coordinate
        \item[$u$:] unknown solution
        \item[$\gamma$:] physics parameters
        \item[$f$:] source data function
        \item[$\mathcal{F}$:] nonlinear differential operator
        \item[$\mathcal{B}$:] boundary/initial condition operator
        \item[$g$:] boundary function
    \end{itemize}

\newpage

    \begin{itemize}
        \item \textbf{Forward problem}: find $u$ for every $z$ (given $\gamma$).
        \item \textbf{Inverse problem}: determine $\gamma$ from data.
        \item \textbf{PINN approximation} [Cuomo et al., JPC 2022]: a neural network $\hat{u}_\theta(z) \approx u(z)$ is trained by minimising:
        \[
            \min_\theta\; \omega_\mathcal{F}\mathcal{L}_\mathcal{F}(\theta) + \omega_\mathcal{B}\mathcal{L}_\mathcal{B}(\theta) + \omega_d\mathcal{L}_{\mathrm{data}}(\theta)
        \]
        where $\mathcal{L}_\mathcal{F}$ enforces the PDE residual, $\mathcal{L}_\mathcal{B}$ enforces boundary/initial conditions, $\mathcal{L}_{\mathrm{data}}$ fits observed data, and $\omega_\mathcal{F},\omega_\mathcal{B},\omega_d$ are loss weights.
        \begin{itemize}
            \item $\mathcal{L}_\mathcal{F}(\theta) = \dfrac{1}{N_c}\displaystyle\sum_{i=1}^{N_c}\|\mathcal{F}(\hat{u}_\theta(z_i)) - f(z_i)\|^2$ \quad (collocation points $\{z_i\}_{i=1}^{N_c}$)
            \item $\mathcal{L}_\mathcal{B}(\theta) = \dfrac{1}{N_B}\displaystyle\sum_{i=1}^{N_B}\|\mathcal{B}(\hat{u}_\theta(z_i)) - g(z_i)\|^2$ \quad (boundary points)
            \item $\mathcal{L}_{\mathrm{data}}(\theta) = \dfrac{1}{N_d}\displaystyle\sum_{i=1}^{N_d}\|\hat{u}_\theta(z_i) - u_i^*\|^2$ \quad (observations $\{(z_i, u_i^*)\}_{i=1}^{N_d}$)
        \end{itemize}
    \end{itemize}

    \item \textbf{Example: PINN for Wave Equation}. 2D acoustic wave equation:
    \[
        \rho\nabla\cdot\!\left(\frac{1}{\rho}\nabla u\right) - \frac{1}{v^2}\frac{\partial^2 u}{\partial t^2} = -\rho\frac{\partial^2 f}{\partial t^2}
    \]
    where $u(t,x)$: pressure (wave-field); $f(t,x)$: point source; $v=\sqrt{\kappa/\rho}$: medium velocity; $\rho$: density; $\kappa$: adiabatic compression modulus. When $\rho$ is constant and $f$ negligible:
    \[
        \nabla^2 u - \frac{1}{v^2}\frac{\partial^2 u}{\partial t^2} = 0
    \]
    \begin{itemize}
        \item \textbf{Model} ($\hat{u}_\theta$): MLP, 10 layers of width 1024, softplus activations (hidden), linear output. Inputs: $x^{(1)}, x^{(2)}, t$; output: $u$.
        \item \textbf{Loss}: $L(\theta) = L_p(\theta) + \lambda L_b(\theta)$, where
        \[
            L_p(\theta) = \frac{1}{N_p}\sum_{i}\left(\left[\nabla^2 - \frac{1}{v(x_i)^2}\frac{\partial^2}{\partial t^2}\right]\hat{u}_\theta\!\left(x_i^{(1)},x_i^{(2)},t_i\right)\right)^{\!2}
        \]
        \[
            L_b(\theta) = \frac{1}{N_b}\sum_{j}\left(\hat{u}_\theta\!\left(x_j^{(1)},x_j^{(2)},t_j\right) - u_{\mathrm{FD}}\!\left(x_j^{(1)},x_j^{(2)},t_j\right)\right)^{\!2}
        \]
        $L_p$: $N_p$ samples with $x_i\in[0,X_{\max}]$, $t_i\in[0,0.2]$;\quad $L_b$: $N_b$ samples with $x_j\in[0,X_{\max}]$, $t_j\in[0,0.02]$.
    \end{itemize}
\end{itemize}

\subsection{AI for Chemistry}
\begin{itemize}
    \item \textbf{Molecular Properties}: Determine a molecule's real-world function, safety, and value (e.g., drug efficacy, material strength).
    \begin{itemize}
        \item \textbf{Typical properties}: Atomization Energy, Free Energy, Heat Capacity, Dipole Moment, Nuclear Magnetic Resonance, Binding Affinity, \ldots
        \item \textbf{Typical methods}:
        \begin{itemize}
            \item Experimental Measurement: slow and expensive.
            \item Quantum Chemistry (e.g., Density Functional Theory (DFT)): computationally intensive.
        \end{itemize}
    \end{itemize}

    \item \textbf{GNN for Molecular Property Prediction}: Input is a molecular graph $G$ with node features $x_v$ and edge features $e_{vw}$.
    \begin{itemize}
        \item \textbf{Message passing} (hidden layers):
        \[
            m_v^{t+1} = \sum_{w \in N(v)} M_t\!\left(h_v^t,\, h_w^t,\, e_{vw}\right), \qquad
            h_v^{t+1} = U_t\!\left(h_v^t,\, m_v^{t+1}\right)
        \]
        where $M_t$ (message) and $U_t$ (update) are learnable functions. \\
        Example:
        $M_t(h_v^t, h_w^t) = c_{vw} h_w^t$, $c_{vw} = (\deg(v)\deg(w))^{-1/2} A_{vw}$;
        $U_t^v(h_v^t, m_v^{t+1}) = \mathrm{ReLU}(W^t m_v^{t+1})$.
        \item \textbf{Output}: $\hat{y} = R\!\left(\{h_v^T \mid v \in G\}\right)$, where $R$ is a readout function.
    \end{itemize}

    \item \textbf{Molecule Graph Generation}: Computationally design novel molecular structures to explore chemical space. Typical deep learning methods:

\section{Lec10: Causal Machine Learning}

\subsection{Introduction}
Goal: infer the effect of treatment $T$ on outcome $Y$ (e.g., treatment on disease, policy on emissions, social media on mental health).

Two scenarios:
\begin{itemize}
    \item \textbf{Observational studies}: cheaper, long-term, but treatment is not randomized — confounding bias exists.
    \item \textbf{Experimental studies (interventions)}: researchers introduce an intervention; subjects are randomized by chance, eliminating confounding.
\end{itemize}

Why randomization is not always possible:
\begin{itemize}
    \item \textbf{Ethical}: e.g., can't randomize people to smoke.
    \item \textbf{Infeasible}: e.g., can't randomize countries into economic systems.
    \item \textbf{Impossible}: e.g., can't change a person's DNA at birth.
\end{itemize}

\subsection{Confounding Variable}
A \textbf{confounding variable} (confounder) influences both the dependent and independent variables, causing a spurious association.
\begin{itemize}
    \item Confounding is a \textbf{causal} concept; it cannot be described in terms of correlations or associations alone.
    \item The existence of confounders is a key reason why \textbf{correlation does not imply causation}.
\end{itemize}

\subsection{Potential Outcomes Framework}
\textbf{Notation}:
\begin{itemize}
    \item $T$: observed treatment; $Y$: observed outcome; $i$: index of a specific unit/individual.
    \item $Y_i(1)$: potential outcome under treatment ($\text{do}(T=1)$).
    \item $Y_i(0)$: potential outcome under no treatment ($\text{do}(T=0)$).
\end{itemize}

\textbf{Individual causal effect}: $Y_i(1) - Y_i(0)$.

\textbf{Average Treatment Effect (ATE)}: expected causal effect of $T$ on $Y$:
\[
    \text{ATE} := \mathbb{E}[Y(1) - Y(0)]
\]
ATE is defined for both experimental and observational studies.

\textbf{Counterfactual problem}: for one subject, only one potential outcome is observable (factual); the other is counterfactual. Hence individual causal effects cannot be directly computed.

Naive estimate of ATE:
\[
    \widehat{\text{ATE}} := \mathbb{E}[Y(1)] - \mathbb{E}[Y(0)]
\]
(人话就是，直接比较治疗组和对照组的平均结果，但是显然这是不准确的) \\
This naive estimate is \textbf{biased} in observational studies due to confounders — the treated and untreated groups may differ systematically (e.g., sicker patients are more likely to choose a specific treatment).

\subsection{Backdoor Adjustment}

When a confounder $C$ affects both $T$ and $Y$, the naive $\mathbb{E}[Y|t]$ conflates causal effect with confounding association. The \textbf{backdoor adjustment} removes this bias by conditioning on $C$ and re-weighting by its marginal distribution:
\[
    \mathbb{E}[Y \mid \mathrm{do}(T=t)] = \mathbb{E}_C\,\mathbb{E}[Y \mid t, C] = \sum_c \mathbb{E}[Y \mid t, c]\, P(c)
\]
Key insight: use \emph{population} weights $P(c)$ (not treatment-specific weights), breaking the spurious path $T \leftarrow C \rightarrow Y$.

\textbf{Example} (Simpson's paradox): Treatment A appears worse overall (16\% vs.\ 19\%), but is better in every stratum after adjusting for severity $C$:
\begin{align*}
    \mathbb{E}[Y \mid \mathrm{do}(A)] &= \tfrac{1450}{2050}(0.15) + \tfrac{600}{2050}(0.30) \approx 0.194 \\
    \mathbb{E}[Y \mid \mathrm{do}(B)] &= \tfrac{1450}{2050}(0.10) + \tfrac{600}{2050}(0.20) \approx 0.129
\end{align*}
So $\text{ATE} = \mathbb{E}[Y|\mathrm{do}(A)] - \mathbb{E}[Y|\mathrm{do}(B)] \approx 0.065 > 0$, confirming A is causally better.

Other methods for ATE estimation: covariate adjustment, propensity scores, instrumental variables.

\subsection{Covariate Adjustment}

Train a regression model $f(x, T)$ on covariates $x = (x_1, \ldots, x_d)$ and treatment $T$ to predict outcome $Y$.

\textbf{Linear model}: assume $Y_t(x) = \beta x + \gamma t + \epsilon_t$, $\mathbb{E}[\epsilon_t] = 0$.

Conditional ATE (CATE):
\begin{align*}
    \text{CATE}(x) &= \mathbb{E}_{p(Y_1|x)}[Y_1] - \mathbb{E}_{p(Y_0|x)}[Y_0] \\
                   &= \mathbb{E}_{\epsilon_0,\epsilon_1}[\beta x + \gamma + \epsilon_1 - \beta x - \epsilon_0] \\
                   &= \gamma + \mathbb{E}[\epsilon_1] - \mathbb{E}[\epsilon_0] = \gamma
\end{align*}

Then $\text{ATE} = \mathbb{E}_{p(x)}[\text{CATE}(x)] = \gamma$, i.e., the coefficient on $T$ directly gives the ATE.

\textbf{Non-linear models}: use random forests or neural networks as $f(x,T)$; estimate CATE$(x) = f(x,1) - f(x,0)$ per individual, then average for ATE.

\subsection{Causal Discovery}

\textbf{Causal Discovery}: given a dataset $\mathcal{D}$ of $d$ variables, learn the causal graph structure over them.

Causal graphs are represented as \textbf{DAGs} (Directed Acyclic Graphs): directed graphs with no directed cycles.

Representative algorithms:
\begin{itemize}
    \item \textbf{PC algorithm} (Spirtes et al., 2001): constraint-based; uses conditional independence tests to orient edges.
    \item \textbf{LiNGAM} (Shimizu et al., 2006): assumes linear non-Gaussian additive noise; uses ICA for identification.
    \item \textbf{CAM} (Bühlmann et al., 2014): score-based with additive noise.
    \item \textbf{NOTEARS} (Zheng et al., 2018): reformulates DAG learning as a continuous optimization with acyclicity constraint $h(W) = \mathrm{tr}(e^{W \circ W}) - d = 0$.
    \item \textbf{DAG-GNN} (Yu et al., 2019), \textbf{GraN-DAG} (Lachapelle et al., 2020): neural network-based structure learning.
    \item \textbf{SCORE} (Rolland et al., 2022): score-function-based causal discovery.
\end{itemize}

\subsection{PC Algorithm (optional)}

The Peter-Clark (PC) algorithm performs causal discovery via conditional independence tests.

\textbf{Steps}:
\begin{enumerate}
    \item Start with a complete undirected graph over all variables.
    \item Remove edge $(A,B)$ if $A \perp B$ (unconditional independence).
    \item For each adjacent pair $(A,B)$ and each neighbor $C$ of $A$ or $B$, remove edge $(A,B)$ if $A \perp B \mid C$.
    \item For each adjacent pair $(A,B)$ and each pair $\{C,D\}$ both adjacent to $A$ or both adjacent to $B$, remove edge $(A,B)$ if $A \perp B \mid \{C,D\}$.
    \item Orient v-structures: if $A - C - B$ with no edge $A$--$B$, orient as $A \to C \leftarrow B$.
    \item Propagate orientations using acyclicity and consistency rules.
\end{enumerate}

\subsection{NOTEARS}

\textbf{Assumption} (Structural Equation Model): graph $G=(V,E)$, $|V|=d$. Let $\pi_j^G$ be the parents of node $j$, $\mathbf{x}_{\pi_j^G}$ the corresponding variables. Each variable satisfies:
\[
    x_j := f_j\!\left(\mathbf{x}_{\pi_j^G}\right) + e_j, \quad e_j \sim \mathcal{N}(0, \sigma_j^2),\ \forall j \in V
\]

\textbf{Key theorem} (Zheng et al., 2018): $\mathbf{W} \in \mathbb{R}^{d\times d}$ is a DAG if and only if
\[
    h(\mathbf{W}) \triangleq \mathrm{tr}\!\left(e^{\mathbf{W} \circ \mathbf{W}}\right) - d = 0, \quad \text{where } e^{\mathbf{M}} \triangleq \sum_{k=0}^{\infty} \frac{\mathbf{M}^k}{k!}
\]
This converts the combinatorial DAG constraint into a smooth equality constraint.

\textbf{Optimization} (linear SEM, i.e.\ $f_j$ linear):
\[
    \min_{\mathbf{W}}\ \frac{1}{2n}\|\mathbf{X} - \mathbf{X}\mathbf{W}\|_F^2 + \lambda\|\mathbf{W}\|_1 \quad \text{s.t. } h(\mathbf{W}) = 0
\]

\begin{itemize}
    \item \textbf{Causal supervised learning} (e.g., IFL, IRM)
    \item Causal generative modeling
    \item Causal explanations
    \item Causal fairness
    \item Causal reinforcement learning
\end{itemize}

\subsection{Invariant Feature Learning (IFL)}

\textbf{Goal}: identify features of $\mathbf{x}$ that are predictive of $y$ across all environments $\mathcal{E}$, and generalize to a new environment $e_{\mathrm{new}}$.

\textbf{Data}: multi-environment datasets $\left\{(\mathbf{x}_i^e, y_i^e)_{i=1}^{N}\right\}_{e \in \mathcal{E}}$ (e.g., data from different hospitals or seasons).

The learned model should generalize to new environment $e_{\mathrm{new}}$. IFL is closely related to transfer learning.

\subsection{Invariant Risk Minimization (IRM)}

IRM (Arjovsky et al., 2019) learns a representation $\phi: \mathcal{X} \to \mathcal{H}$ and an invariant classifier $w: \mathcal{H} \to \mathcal{Y}$ by minimizing the total risk across all environments, subject to $w$ being simultaneously optimal for every environment:
\[
    \min_{\phi,\, w} \sum_{e \in \mathcal{E}} \mathcal{R}^e(w \circ \phi) \quad \text{s.t.} \quad w \in \operatorname*{arg\,min}_{\bar{w}}\ \mathcal{R}^e(\bar{w} \circ \phi) \quad \forall e \in \mathcal{E}
\]
where the per-environment risk is $\mathcal{R}^e(f) := \mathbb{E}_{\mathbf{x}^e, y^e}[\ell(f(\mathbf{x}^e), y^e)]$.

This is a bi-level optimization problem and is highly non-convex. The practical relaxation penalizes the gradient norm:
\[
    \min_{\phi,\, w} \sum_{e \in \mathcal{E}} \left[\mathcal{R}^e(w \circ \phi) + \lambda \left\|\nabla_w \mathcal{R}^e(w \circ \phi)\right\|^2\right]
\]

\section{Lec11: DP and FL}
\subsection{Privacy Protection Strategies}
\begin{itemize}
    \item De-identify data by removing/modifying personally identifiable information.
    \item Release only aggregate statistics.
    \item Design randomized algorithms (e.g., differential privacy).
    \item Train ML models on-device without centralizing raw data (federated learning).
\end{itemize}

\subsection{Privacy Attacks}
\begin{itemize}
    \item \textbf{Membership inference attack}: determine whether a sample $\mathbf{x}$ was in the training set $\mathcal{D}$; often detectable from model prediction confidence alone.
    \item \textbf{Unintended memorization attack}: query a language model to recover private text it memorized during training (e.g., fill in masked sensitive fields).
    \item \textbf{Model inversion attack}: recover information about training data from the trained model's outputs. Even releasing only model weights may leak sensitive information.
    \item \textbf{Reconstruction attack} (Dinur \& Nissim, 2003): reconstruct private values in database $X$ from query results $q(X)$. E.g., querying ``how many rows satisfying (identifier conditions) have disease $= 1$?'' with enough queries can recover individual records. This attack motivates differential privacy.
    \item \textbf{Differencing attack}: with side information, even a single aggregate statistic can reveal an individual's sensitive attribute.
\end{itemize}

\textbf{Why naive strategies fail}:
\begin{itemize}
    \item Releasing data with PII removed $\Rightarrow$ model inversion / reconstruction attacks still work.
    \item Releasing aggregate statistics $\Rightarrow$ differencing and reconstruction attacks.
    \item Releasing model weights $\Rightarrow$ membership inference and unintended memorization.
\end{itemize}

\subsection{Randomized Response}

A survey technique that adds randomness to responses to protect individual privacy while preserving population statistics.

\textbf{Protocol} (binary response): flip a coin; if Heads, answer truthfully; if Tails, flip again and answer Yes/No with equal probability. This gives:
\[
    \Pr(\text{Yes} \mid \text{Dodge}) = \tfrac{3}{4}, \quad \Pr(\text{Yes} \mid \text{NoDodge}) = \tfrac{1}{4}
\]

\textbf{Privacy intuition}: given a ``Yes'' answer and prior $\Pr(\text{Dodge})=0.02$, the posterior is only $\approx 0.058$ — beliefs shift little, so the individual is protected.

\textbf{Unbiased estimation}: let $X_\mathrm{T}^{(i)}$ be individual $i$'s true response and $X_\mathrm{R}^{(i)}$ be their randomized response. The truthful MLE is $\hat{\mu}_\mathrm{T} = \frac{1}{N}\sum_{i=1}^N X_\mathrm{T}^{(i)}$. Since
\[
    \mathbb{E}\!\left[X_\mathrm{R}^{(i)}\right] = \tfrac{1}{4}(1-\mu_i) + \tfrac{3}{4}\mu_i
\]
the unbiased estimator from randomized responses is:
\[
    \hat{\mu}_\mathrm{R} = \frac{2}{N}\sum_i X_\mathrm{R}^{(i)} - \frac{1}{2}
\]

The cost of RR is \textbf{higher variance and lower accuracy} ($\mathrm{Var}$ increases 4$\times$, computational complexity is unaffected) — this is the fundamental \textbf{privacy-utility tradeoff}, motivating Differential Privacy. \\
(人话就是，保护隐私的代价是准确率降低了，因为随机化增加了噪声，所以方差也增加了，但是不影响计算复杂度)
\\
Truthful MLE variance:
\[
    \mathrm{Var}(\hat{\mu}_\mathrm{T}) = \frac{1}{N}\mathrm{Var}\!\left(X_\mathrm{T}^{(i)}\right) = \frac{1}{N}\mu(1-\mu)
\]
RR estimator variance (the inequality holds because randomization adds noise: $\mathrm{Var}(X_\mathrm{R}^{(i)}) \geq \mathrm{Var}(X_\mathrm{T}^{(i)})$):
\[
    \mathrm{Var}(\hat{\mu}_\mathrm{R}) = \frac{4}{N}\mathrm{Var}\!\left(X_\mathrm{R}^{(i)}\right) \geq \frac{4}{N}\mathrm{Var}\!\left(X_\mathrm{T}^{(i)}\right) = 4\,\mathrm{Var}(\hat{\mu}_\mathrm{T})
\]

\subsection{Differential Privacy (DP，差分隐私)}

Three components:
\begin{itemize}
    \item \textbf{Database} $\mathcal{D}$: contains sensitive individual information.
    \item \textbf{Curator} (trusted): holds $\mathcal{D}$ and responds to queries via a \textbf{mechanism}.
    \item \textbf{Analyst}: submits queries; receives noisy responses.
\end{itemize}

\textbf{Neighbouring datasets} $\mathcal{D}_1, \mathcal{D}_2$: two databases differing in exactly one entry.

\textbf{Core idea}: if the mechanism's output distribution is nearly identical on $\mathcal{D}_1$ and $\mathcal{D}_2$, an attacker cannot determine which database was used, thus learning little about any individual.

\begin{mdframed}[linewidth=1.5pt, linecolor=blue!60, backgroundcolor=blue!5,
    frametitle={\textbf{Definition 1 ($\varepsilon$-DP, Original)}},
    frametitlebackgroundcolor=blue!40, frametitlerule=true]
A mechanism $\mathcal{M}$ is \textbf{$\varepsilon$-differentially private} if for any two neighbouring databases $\mathcal{D}_1, \mathcal{D}_2$ and any set $\mathcal{R}$ of possible responses:
\[
    \Pr\!\left(\mathcal{M}(\mathcal{D}_1) \in \mathcal{R}\right) \leq \exp(\varepsilon)\,\Pr\!\left(\mathcal{M}(\mathcal{D}_2) \in \mathcal{R}\right)
\]
Equivalently, for any response $y$:
\[
    \exp(-\varepsilon) \leq \frac{\Pr(\mathcal{M}(\mathcal{D}_1) = y)}{\Pr(\mathcal{M}(\mathcal{D}_2) = y)} \leq \exp(\varepsilon)
\]
\end{mdframed}
人话：无论你在不在数据库里，机制的输出分布最多相差 $e^\varepsilon$ 倍——攻击者无法从输出中判断你的数据是否被使用。$\varepsilon$ 越小隐私越强，$\varepsilon=0$ 时两个数据库完全不可区分。

\begin{mdframed}[linewidth=1.5pt, linecolor=blue!60, backgroundcolor=blue!5,
    frametitle={\textbf{Definition 2 ($(\varepsilon,\delta)$-DP, Relaxed)}},
    frametitlebackgroundcolor=blue!40, frametitlerule=true]
A mechanism $\mathcal{M}$ is \textbf{$(\varepsilon,\delta)$-differentially private} if for any two neighbouring databases $\mathcal{D}_1, \mathcal{D}_2$ and any set $\mathcal{R}$:
\[
    \Pr\!\left(\mathcal{M}(\mathcal{D}_1) \in \mathcal{R}\right) \leq \exp(\varepsilon)\,\Pr\!\left(\mathcal{M}(\mathcal{D}_2) \in \mathcal{R}\right) + \delta
\]
where $\varepsilon, \delta \geq 0$ and $\delta$ is small, decreasing with $n$.
\end{mdframed}
人话：在 $\varepsilon$-DP 基础上允许以最多 $\delta$ 的概率"破防"——相当于以极小概率发生隐私泄露，换来更大的算法设计灵活性（如 Gaussian 机制）。

\subsection{Properties of DP}

\textbf{Property 1 (Posterior bound)}: If $\mathcal{M}$ is $\varepsilon$-DP and an attacker's prior on $x \in \mathcal{D}$ is $p$, then after observing any output $y = \mathcal{M}(\mathcal{D})$:
\[
    p \cdot e^{-\varepsilon} \leq \Pr(x \in \mathcal{D} \mid y) \leq p \cdot e^{\varepsilon}
\]
人话：DP 从根本上限制了攻击者能从输出中学到多少——后验概率被夹在先验的 $e^{\pm\varepsilon}$ 倍之间，无法大幅更新。

\textbf{Property 2 (Post-processing)}: If $\mathcal{M}$ is $\varepsilon$-DP and $\mathcal{A}$ is any algorithm that does not access $\mathcal{D}$ directly, then $\mathcal{A} \circ \mathcal{M}$ is also $\varepsilon$-DP. Proof: let $\mathcal{Y} = \{y : \mathcal{A}(y) \in \mathcal{O}\}$ be the pre-image of output set $\mathcal{O}$ under $\mathcal{A}$, then
\begin{align*}
    \Pr(\mathcal{A}(\mathcal{M}(\mathcal{D}_1)) \in \mathcal{O})
    &= \Pr(\mathcal{M}(\mathcal{D}_1) \in \mathcal{Y}) \\
    &\leq \exp(\varepsilon)\,\Pr(\mathcal{M}(\mathcal{D}_2) \in \mathcal{Y}) \\
    &= \exp(\varepsilon)\,\Pr(\mathcal{A}(\mathcal{M}(\mathcal{D}_2)) \in \mathcal{O})
\end{align*}
人话：对 DP 机制的输出做任意后处理（分析、发布、再计算），隐私保证不会变差。

\subsection{Pros and Cons of DP}

\textbf{Advantages}:
\begin{itemize}
    \item Rigorous mathematical guarantee; makes no assumptions on the adversary or database.
    \item Interpretable, quantifiable ($\varepsilon$), and composable formalism.
    \item De-facto standard in privacy research and industry.
\end{itemize}

\textbf{Disadvantages}:
\begin{itemize}
    \item Can be too restrictive; various relaxations (e.g., $(\varepsilon,\delta)$-DP, Rényi DP) have been proposed.
    \item Other complementary techniques exist (e.g., homomorphic encryption).
\end{itemize}

\subsection{Laplace Mechanism}

Add Laplace noise to the query response. The Laplace distribution:
\[
    p(y;\, \mu, b) = \frac{1}{2b} \exp\!\left(-\frac{|y - \mu|}{b}\right)
\]

\begin{mdframed}[linewidth=1.5pt, linecolor=blue!60, backgroundcolor=blue!5,
    frametitle={\textbf{Definition 3 (Global $\ell_1$ sensitivity)}},
    frametitlebackgroundcolor=blue!40, frametitlerule=true]
For neighbouring datasets $\mathcal{D}, \mathcal{D}'$, the global $\ell_1$ sensitivity of a query $f: \mathbb{N}^{|\mathcal{X}|} \to \mathbb{R}^K$ is:
\[
    \Delta f = \max_{\mathcal{D},\mathcal{D}'} \left\| f(\mathcal{D}) - f(\mathcal{D}') \right\|_1
\]
\end{mdframed}
人话：$\Delta f$ 衡量单个人的加入/离开最多能让查询结果变化多少，决定了需要加多大的噪声。

The scale parameter is set to $b = \Delta f / \varepsilon$. Example: if $f$ counts individuals in $k$ disjoint buckets, then $\Delta f = 1$ (one person changes exactly one bucket count by 1).

\textbf{Laplace Mechanism:} Output $y = f(\mathcal{D}) + z$, where each $z_i \sim \mathrm{Lap}(\Delta f/\varepsilon)$ i.i.d., i.e.,
\[
    y_i \sim \mathrm{Laplace}\!\left(f(\mathcal{D})_i,\, \frac{\Delta f}{\varepsilon}\right), \quad i = 1, 2, \ldots, K.
\]

\begin{mdframed}[linewidth=1.5pt, linecolor=blue!60, backgroundcolor=blue!5,
    frametitle={\textbf{Theorem 1 (DP guarantees for Laplace mechanism)}},
    frametitlebackgroundcolor=blue!40, frametitlerule=true]
\textit{Let $\varepsilon > 0$ and $f: \mathbb{N}^{|\mathcal{X}|} \to \mathbb{R}^K$. The Laplace mechanism $\mathcal{M}_{\mathrm{Lap}}(\cdot, f, \varepsilon): \mathbb{N}^{|\mathcal{X}|} \to \mathcal{O}$ satisfies $\varepsilon$-DP.}
\end{mdframed}

\textbf{Proof:}
\begin{align*}
\frac{p(y \mid \mathcal{D}_1)}{p(y \mid \mathcal{D}_2)}
&= \frac{\prod_{i=1}^{K} \frac{\varepsilon}{2\Delta f} \exp\!\left(-\dfrac{\varepsilon\,|f(\mathcal{D}_1)_i - y_i|}{\Delta f}\right)}
        {\prod_{i=1}^{K} \frac{\varepsilon}{2\Delta f} \exp\!\left(-\dfrac{\varepsilon\,|f(\mathcal{D}_2)_i - y_i|}{\Delta f}\right)} \\[6pt]
&= \prod_{i=1}^{K} \exp\!\left(\frac{\varepsilon\bigl(|f(\mathcal{D}_2)_i - y_i| - |f(\mathcal{D}_1)_i - y_i|\bigr)}{\Delta f}\right) \\[6pt]
&\leq \prod_{i=1}^{K} \exp\!\left(\frac{\varepsilon\,|f(\mathcal{D}_2)_i - f(\mathcal{D}_1)_i|}{\Delta f}\right)
 = \exp\!\left(\frac{\varepsilon \sum_{i=1}^{K}|f(\mathcal{D}_2)_i - f(\mathcal{D}_1)_i|}{\Delta f}\right) \\[6pt]
&= \exp\!\left(\frac{\varepsilon\,\|f(\mathcal{D}_2) - f(\mathcal{D}_1)\|_1}{\Delta f}\right)
 \leq \exp(\varepsilon) \qquad \blacksquare
\end{align*}

\subsection{Exponential Mechanism}

Laplace/Gaussian mechanisms only work for \textbf{numeric queries} and require a sufficiently regular utility function. For non-numeric or discrete output spaces, we need the Exponential Mechanism.

Consider queries $f: \mathbb{N}^{|\mathcal{X}|} \to \mathcal{O}$ with an abstract output space $\mathcal{O}$, e.g.:
\begin{itemize}
    \item $\mathcal{O} = \{$`Google', `GitHub', `Jingdong', \ldots$\}$ (websites)
    \item $\mathcal{O} = \{$`dark', `blond', `white', `brown', `red'$\}$ (hair color)
\end{itemize}

Define a \textbf{score (utility) function}:
\[
    u: \mathbb{N}^{|\mathcal{X}|} \times \mathcal{O} \to \mathbb{R}
\]
where $u(\mathcal{D}, o)$ measures how good it is to return $o$ on dataset $\mathcal{D}$. $u$ can be \textbf{arbitrary} and should be designed according to the use-case. The true answer $o = f(\mathcal{D})$ is typically assigned the maximum score.

\begin{mdframed}[linewidth=1.5pt, linecolor=blue!60, backgroundcolor=blue!5,
    frametitle={\textbf{Definition 4 (Sensitivity of score function)}},
    frametitlebackgroundcolor=blue!40, frametitlerule=true]
Let $\mathcal{D}$ and $\mathcal{D}'$ be two neighbouring datasets. The sensitivity of $u: \mathbb{N}^{|\mathcal{X}|} \times \mathcal{O} \to \mathbb{R}$ is:
\[
    \Delta u = \max_{o \in \mathcal{O}}\, \max_{\mathcal{D},\mathcal{D}'} \left| u(\mathcal{D}, o) - u(\mathcal{D}', o) \right|
\]
\end{mdframed}
人话：$\Delta u$ 衡量换一个相邻数据集时 score 最多变多少，决定指数分布里加多大的"模糊度"。

\textbf{Exponential Mechanism} $\mathcal{M}_{\mathrm{Exp}}\!\left(\mathcal{D},\, f{:}\,\mathbb{N}^{|\mathcal{X}|}\to\mathcal{O},\, u{:}\,\mathbb{N}^{|\mathcal{X}|}\times\mathcal{O}\to\mathbb{R},\, \varepsilon\right)$: output $o \in \mathcal{O}$ with probability
\[
    \Pr[o] = \frac{\exp\!\left(\dfrac{u(\mathcal{D},o)\cdot\varepsilon}{2\Delta u}\right)}{\displaystyle\sum_{o'\in\mathcal{O}} \exp\!\left(\dfrac{u(\mathcal{D},o')\cdot\varepsilon}{2\Delta u}\right)}
\]
The distribution is related to Boltzmann distributions (cf.\ softmax regression).

\begin{mdframed}[linewidth=1.5pt, linecolor=blue!60, backgroundcolor=blue!5,
    frametitle={\textbf{Theorem 2 (DP guarantees for Exponential Mechanism)}},
    frametitlebackgroundcolor=blue!40, frametitlerule=true]
\textit{Let $\varepsilon > 0$, $f: \mathbb{N}^{|\mathcal{X}|} \to \mathcal{O}$ and $u: \mathbb{N}^{|\mathcal{X}|} \times \mathcal{O} \to \mathbb{R}$. $\mathcal{M}_{\mathrm{Exp}}(\cdot, f, u, \varepsilon)$ satisfies $\varepsilon$-DP.}
\end{mdframed}
人话：只要按上面的概率公式采样输出，就自动满足 $\varepsilon$-差分隐私。

\textbf{Proof:}
\begin{align*}
&\frac{\Pr[\mathcal{M}_{\mathrm{Exp}}(\mathcal{D}_1,f,u,\varepsilon)=o]}{\Pr[\mathcal{M}_{\mathrm{Exp}}(\mathcal{D}_2,f,u,\varepsilon)=o]}
= \frac{\exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_1,o)}{2\Delta u}\right)}{\exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_2,o)}{2\Delta u}\right)}
\cdot \frac{\displaystyle\sum_{o'\in\mathcal{O}}\exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_2,o')}{2\Delta u}\right)}{\displaystyle\sum_{o'\in\mathcal{O}}\exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_1,o')}{2\Delta u}\right)} \\[8pt]
&= \exp\!\left(\frac{\varepsilon\bigl(u(\mathcal{D}_1,o)-u(\mathcal{D}_2,o)\bigr)}{2\Delta u}\right)
\cdot \frac{\displaystyle\sum_{o'\in\mathcal{O}}\exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_2,o')}{2\Delta u}\right)}{\displaystyle\sum_{o'\in\mathcal{O}}\exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_1,o')}{2\Delta u}\right)} \\[8pt]
&\leq \exp\!\left(\frac{\varepsilon}{2}\right)
\cdot \frac{\displaystyle\sum_{o'\in\mathcal{O}}\exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_2,o')}{2\Delta u}\right) / \exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_1,o')}{2\Delta u}\right) \cdot \exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_1,o')}{2\Delta u}\right)}{\displaystyle\sum_{o'\in\mathcal{O}}\exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_1,o')}{2\Delta u}\right)} \\[8pt]
&\leq \exp\!\left(\frac{\varepsilon}{2}\right) \cdot \exp\!\left(\frac{\varepsilon}{2}\right)
\cdot \frac{\displaystyle\sum_{o'\in\mathcal{O}}\exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_1,o')}{2\Delta u}\right)}{\displaystyle\sum_{o'\in\mathcal{O}}\exp\!\left(\dfrac{\varepsilon u(\mathcal{D}_1,o')}{2\Delta u}\right)}
= \exp(\varepsilon) \qquad \blacksquare
\end{align*}

\subsection{Federated Learning (FL，联邦学习)}

\textbf{Federated learning} (collaborative learning): trains a model across multiple independent sessions, each using its own local dataset — raw data never leaves the device.

\subsubsection{Objective Function}
\[
    f(\Theta; \mathcal{D}) = \frac{1}{K} \sum_{k=1}^{K} f_k(\theta_k;\, \mathcal{D}_k)
\]
where $K$ is the number of clients, $\mathcal{D} = \{\mathcal{D}_1, \cdots, \mathcal{D}_K\}$ with $\mathcal{D}_k$ the local dataset of client $k$,
$\Theta = \{\theta_1, \cdots, \theta_K\}$ with $\theta_k$ the local model parameters, and $f_k$ the local objective of client $k$.

Goal: optimize $f(\Theta;\mathcal{D})$ such that $\theta_1, \cdots, \theta_K$ converge to a common $\theta$.

\subsubsection{Main Steps (repeat until convergence)}
\begin{enumerate}
    \item Server selects a random subset of clients and sends them the current global model weights.
    \item Each selected client computes an updated model using its local data.
    \item Clients send model updates back to the server.
    \item Server aggregates the updates (typically by averaging) to construct an improved global model.
\end{enumerate}

\subsubsection{Normal Assumptions}
\begin{itemize}
    \item \textbf{Distributed storage (non-IID):} data is localized to each user's own usage, hard to be representative of the population.
    \item \textbf{Heterogeneous services (unbalanced):} some users rely much more heavily on particular services than others.
    \item \textbf{Massively distributed computing:} a large number of devices may be updated simultaneously.
    \item \textbf{Limited communication:} mobile devices are frequently offline or on slow/expensive connections.
\end{itemize}

\subsubsection{Architectures}
\begin{itemize}
    \item \textbf{Centralized FL:} a central server orchestrates all steps — node selection and model aggregation. Server may become a bottleneck.
    \item \textbf{Decentralized FL:} no central server; nodes coordinate themselves. Network topology may affect performance.
    \item \textbf{Heterogeneous FL:} a combination of the above two.
\end{itemize}

\subsubsection{FedSGD vs.\ FedAvg}

\textbf{FedSGD} (Federated SGD): each client computes local gradient $g_k = \nabla F_k(w_t)$; server averages them weighted by local dataset size and performs one GD step:
\[
    w_{t+1} \leftarrow w_t - \eta \sum_{k=1}^{K} \frac{n_k}{n}\, g_k
\]

\textbf{FedAvg} (Federated Averaging): each client performs multiple local batch updates, then exchanges updated weights rather than gradients:
\[
    w_{t+1}^k \leftarrow w_t - \eta g_k, \qquad
    w_{t+1} \leftarrow \sum_{k=1}^{K} \frac{n_k}{n}\, w_{t+1}^k
\]
If all local nodes start from the same initialization, averaging gradients is strictly equivalent to averaging weights.
(FedAvg 更常用，因为通信效率更高，尤其在网络不稳定的情况下)
\subsubsection{FL and Differential Privacy}
FL does \textbf{not} automatically satisfy differential privacy, but local updates can be randomized (e.g., via Gaussian mechanism) to make them differentially private.

人话：FL 本身不保证隐私，数据不出门 $\neq$ 隐私安全，还需要在梯度上加噪声才行。


\section{Lec12: Fairness in Machine Learning}
\subsection{Motivation}

ML provides automated decision making.
\begin{itemize}
    \item \textbf{Ideal:} cheaper, more accurate, more impartial decisions that improve our lives.
    \item \textbf{Reality:} if not carefully designed, automated decisions can encode bias, harm people, and make lives worse.
\end{itemize}

\subsection{Definition of Fairness}

Common fairness definitions: Fairness Through Unawareness, Fairness Through Awareness, Demographic Parity, Equal Opportunity, Equalized Odds, Treatment Equality, Test Fairness, Counterfactual Fairness, Fairness in Relational Domains, Conditional Statistical Parity.

\begin{itemize}
    \item \textbf{Fairness Through Unawareness:} an algorithm is fair as long as protected attributes are not explicitly used in decision-making. (Default approach in ML.)

    \textit{Limitations:}
    \begin{itemize}
        \item masking sensitive features is hard in practice;
        \item fails when unprotected attributes are correlated with protected ones.
    \end{itemize}
    \item \textbf{Fairness Through Awareness:} an algorithm is fair if it gives similar predictions to similar individuals.

    \item \textbf{Demographic Parity} (a.k.a.\ statistical parity): predictor $\hat{Y}$ satisfies demographic parity if
    \[
        P(\hat{Y} \mid A=0) = P(\hat{Y} \mid A=1)
    \]
    where $A$ is the protected feature. The likelihood of a positive prediction should be equal across groups.

    \textit{Issues:} independent of the target variable — when suitable candidate fractions differ across groups, demographic parity rejects the optimal classifier and may disadvantage qualified individuals.

    \textit{When to use:} when correcting historical bias in data is the goal (e.g., data where nearly no women were hired).

    \item \textbf{Equal Opportunity:} binary predictor $\hat{Y}$ satisfies equal opportunity w.r.t.\ $A$ and $Y$ if
    \[
        P(\hat{Y}=1 \mid A=0,\, Y=1) = P(\hat{Y}=1 \mid A=1,\, Y=1)
    \]
    i.e., TPR is equal across groups. Weaker version of Equalized Odds.

    \item \textbf{Equalized Odds:} $\hat{Y}$ satisfies equalized odds w.r.t.\ $A$ and $Y$ if $\hat{Y} \perp A \mid Y$, i.e.,
    \[
        P(\hat{Y}=1 \mid A=0,\, Y=y) = P(\hat{Y}=1 \mid A=1,\, Y=y), \quad y \in \{0,1\}
    \]
    i.e., both TPR and FPR are equal across groups.
\end{itemize}
(这又几把是什么)

\subsection{Categorization of Fairness Definitions}
\begin{itemize}
    \item \textbf{Individual Fairness:} give similar predictions to similar individuals. Requires a similarity metric, which is generally hard to find in practice.
    \item \textbf{Group Fairness} (statistical parity): treat different groups equally. Does not account for individual merits; may favor less qualified group members.
    \item \textbf{Subgroup Fairness:} combines the best properties of individual and group fairness notions.
\end{itemize}

\subsection{Impossibility Theorem of Fairness}

\textbf{No Free Lunch in Fairness:} it is impossible to simultaneously satisfy all group fairness definitions (demographic parity, equalized odds, predictive rate parity) except in degenerate cases.

The three major group fairness criteria are \textbf{pairwise incompatible}: satisfying any one generally violates the others. Concretely:
\begin{itemize}
    \item If base rates differ across groups, i.e., $P(Y=1 \mid A=0) \neq P(Y=1 \mid A=1)$, then demographic parity and equalized odds cannot both hold for a non-trivial classifier.
    \item A classifier satisfying equalized odds will typically violate predictive rate parity (calibration within groups), and vice versa.
\end{itemize}

There is no single ``correct'' definition of fairness. Data scientists must consciously choose which criterion to optimize, based on the context and potential harms of the application.

人话：公平的各种定义之间互相矛盾，不可能同时满足，只能根据具体场景选一个来优化。

\subsection{Fairness Intervention Types}

Three stages to intervene for fairness:
\begin{itemize}
    \item \textbf{Pre-processing:} modify the training data to remove bias before training.
    \item \textbf{In-processing:} incorporate fairness constraints directly into the learning objective. Example:
    \[
        \min_{\theta}\ \mathcal{L}(\theta) \quad \text{s.t.} \quad \left|P(\hat{Y} \neq Y \mid A=a) - P(\hat{Y} \neq Y \mid A=b)\right| \leq \epsilon
    \]
    \item \textbf{Post-processing:} adjust the model's predictions after training to satisfy fairness criteria.
\end{itemize}

\subsection{Learning Fair Representations}

Goal: find a representation $Z$ of input $X$ that removes any information about the sensitive attribute $S$, so that vendors can use $Z$ to predict $Y$ freely with fairness guaranteed.

Idea: separate responsibilities — a trusted \textit{society} encodes $X$ into a fair $Z$; an untrusted \textit{vendor} uses $Z$ to make predictions $Y$ without access to sensitive attributes.

\subsubsection{Notation}
\begin{itemize}
    \item $X$: entire dataset of individuals; each $\mathbf{x} \in X$ is a vector of length $D$ (attributes of a person).
    \item $S$: sensitive attribute, $S \in \{0, 1\}$.
    \item $Z$: multinomial random variable with $K$ values, each representing an intermediate ``prototype''. Each prototype $k$ is associated with a vector $\mathbf{v}_k$ in the same space as $\mathbf{x}$.
\end{itemize}

\subsubsection{Mapping to Prototypes}
Softmax over distance to each prototype:
\[
    P(Z=k \mid \mathbf{x}) = \frac{\exp(-d(\mathbf{x}, \mathbf{v}_k))}{\sum_{j=1}^{K} \exp(-d(\mathbf{x}, \mathbf{v}_j))}
\]
Denote $M_{n,k} = P(Z=k \mid \mathbf{x}_n)$.

\subsubsection{Fairness Constraint and Discrimination Loss}
Statistical parity requires $M_k^+ = M_k^-,\ \forall k$, where
\[
    M_k^+ = \mathbb{E}_{\mathbf{x} \in X^+} P(Z=k \mid \mathbf{x}) = \frac{1}{|X_0^+|} \sum_{n \in X_0^+} M_{n,k}
\]
and $M_k^-$ is defined analogously for $X^-$. This is enforced via the discrimination loss:
\[
    \mathcal{L}_{\mathrm{discrim}} = \frac{1}{K} \sum_{k=1}^{K} \left|
        \frac{1}{N_0} \sum_{i:\, s^{(i)}=0} \Pr(Z=k \mid \mathbf{x}^{(i)})
        - \frac{1}{N_1} \sum_{i:\, s^{(i)}=1} \Pr(Z=k \mid \mathbf{x}^{(i)})
    \right|
\]
\textit{Proof that $\mathcal{L}_{\mathrm{discrim}} = 0 \Rightarrow$ demographic parity:}
\begin{align*}
\Pr(Y=1 \mid s^{(i)}=1)
&= \frac{1}{N_1}\sum_{i:\,s^{(i)}=1}\sum_{k=1}^{K} \Pr(Z=k \mid \mathbf{x}^{(i)})\,\Pr(Y=1 \mid Z=k) \\
&= \sum_{k=1}^{K} \left[\frac{1}{N_1}\sum_{i:\,s^{(i)}=1} \Pr(Z=k \mid \mathbf{x}^{(i)})\right]\Pr(Y=1 \mid Z=k) \\
&= \sum_{k=1}^{K} \left[\frac{1}{N_0}\sum_{i:\,s^{(i)}=0} \Pr(Z=k \mid \mathbf{x}^{(i)})\right]\Pr(Y=1 \mid Z=k) \\
&= \Pr(Y=1 \mid s^{(i)}=0)
\end{align*}

\subsubsection{Reconstruction ($Z \to \mathbf{x}$)}
Retains information about $X$ by penalizing reconstruction error:
\[
    \hat{\mathbf{x}}_n = \sum_{k=1}^{K} M_{n,k}\, \mathbf{v}_k, \qquad
    \mathcal{L}_{\mathrm{reconst}} = \frac{1}{N}\sum_{i=1}^{N} \left\|\mathbf{x}^{(i)} - \hat{\mathbf{x}}^{(i)}\right\|^2
\]

\subsubsection{Prediction ($Z \to \hat{y}$)}
\[
    \hat{y}_n = \sum_{k=1}^{K} M_{n,k}\, w_k, \qquad
    \mathcal{L}_{\mathrm{pred}} = \frac{1}{N}\sum_{i=1}^{N}\left(-y^{(i)}\log\hat{y}^{(i)} - (1-y^{(i)})\log(1-\hat{y}^{(i)})\right)
\]

\subsubsection{Overall Objective}
\[
    \mathcal{L}(\{\mathbf{v}_k\}, \mathbf{w}) = \lambda_1\,\mathcal{L}_{\mathrm{pred}} + \lambda_2\,\mathcal{L}_{\mathrm{reconst}} + \lambda_3\,\mathcal{L}_{\mathrm{discrim}}
\]

\subsubsection{Experiments}
\textbf{Datasets:}
\begin{itemize}
    \item \textbf{German Credit:} classify credit risk (good/bad); sensitive feature: Age.
    \item \textbf{Adult Income:} 45,222 instances, 14 attributes; predict annual income $>50$K; sensitive feature: Gender.
    \item \textbf{Heritage Health:} 147,473 instances, 139 attributes; predict hospital stay; sensitive feature: Age.
\end{itemize}

\textbf{Evaluation metrics:}
\begin{itemize}
    \item Classification accuracy:
    \[
        \mathrm{Acc} = 1 - \frac{1}{N}\sum_{n=1}^{N} |y_n - \hat{y}_n|
    \]
    \item Discrimination (gap in positive prediction rate between groups):
    \[
        \mathrm{Discrim} = \left| \frac{\sum_{i:\,s^{(i)}=1} \hat{y}^{(i)}}{N_1} - \frac{\sum_{i:\,s^{(i)}=0} \hat{y}^{(i)}}{N_0} \right|
    \]
\end{itemize}

\subsection{Fair VAE}

Discrete $Z$ based on prototypes is too limiting. Instead, learn a continuous representation with a new fairness criterion: ensure $p(\mathbf{z} \mid s)$ is indistinguishable across different values of $s$.

\subsubsection{Maximum Mean Discrepancy (MMD)}

MMD is a quantitative measure of distance between two distributions. Given a feature map $\psi$:
\[
    \mathrm{MMD}(p;\, q) = \left\|\mathbb{E}_{\mathbf{z} \sim p}[\psi(\mathbf{z})] - \mathbb{E}_{\mathbf{z} \sim q}[\psi(\mathbf{z})]\right\|^2
\]
If $\psi$ is sufficiently expressive, $\mathrm{MMD}(p;q) = 0$ iff $p = q$ (made precise via kernels).

\subsubsection{VFAE Objective}

Train a VAE with the constraint that the MMD between $p(\mathbf{z} \mid s=0)$ and $p(\mathbf{z} \mid s=1)$ is small \textit{[Louizos et al. 2016]}:
\begin{itemize}
    \item Encoder maps input $\mathbf{x}$ to latent code $\mathbf{Z}$ (with MMD constraint).
    \item Decoder reconstructs $\tilde{\mathbf{x}}$ from $\mathbf{Z}$ and $s$ (decoder receives $s$ since reconstruction would otherwise be impossible).
    \item Prediction $Y$ is made from $\mathbf{Z}$ only, without access to $s$.
\end{itemize}
\[
    \mathcal{L}_{\mathrm{VFAE}}(\phi, \theta;\, \mathbf{X}) = \mathcal{L}_{\mathrm{VAE}}(\phi, \theta;\, \mathbf{X}) - \beta\,\ell_{\mathrm{MMD}}\!\left(\mathbf{Z}_{\delta\{s=0\}},\, \mathbf{Z}_{\delta\{s=1\}}\right)
\]

\subsection{Flexibly Fair VAE (FFVAE)}

\subsubsection{Notation}

The latent code is split into two subspaces:
\begin{itemize}
    \item $x \in \mathcal{X}$: non-sensitive observations (e.g., pixel values or tabular features).
    \item $a \in \{0,1\}^{N_a}$: vector of binary sensitive attributes.
    \item $z \in \mathbb{R}^{N_z}$: non-sensitive subspace of the latent code.
    \item $b \in \mathbb{R}^{N_b}$: sensitive subspace of the latent code.
\end{itemize}

\subsubsection{Objectives}

Original VAE objective (with sensitive attributes included):
\[
    L_{\mathrm{VAE}}(p,q) = \mathbb{E}_{q(z,b\mid x,a)}\!\left[\log p(x,a \mid z,b)\right] - D_{KL}\!\left[q(z,b\mid x,a)\,\|\,p(z,b)\right]
\]

Flexibly Fair VAE objective \textit{[Creager et al. 2019]}, which additionally encourages $z \perp b$ (low mutual information) and sensitive attribute predictability from $b$:
\begin{align*}
    L_{\mathrm{FFVAE}}(p,q) &= \mathbb{E}_{q(z,b\mid x)}\!\left[\log p(x \mid z,b) + \alpha \log p(a \mid b)\right] \\
    &\quad - \gamma\, D_{KL}\!\left(q(z,b)\,\Big\|\,q(z)\prod_{j} q(b_j)\right) \\
    &\quad - D_{KL}\!\left[q(z,b\mid x)\,\|\,p(z,b)\right]
\end{align*}

\subsubsection{Applications}
\begin{itemize}
    \item \textbf{Fair Classification:} use $z$ (free of sensitive info) for downstream prediction.
    \item \textbf{Predictiveness:} train a classifier to predict sensitive attribute $a_i$ from $b_i$ alone — verifies $b$ captures sensitive information.
    \item \textbf{Disentanglement:} train a classifier to predict $a_i$ from representations with $b_i$ removed — verifies $z$ is free of sensitive information.
\end{itemize}

\section{Lec13: Explanable AI (XAI)}
\subsection{Motivation}

Explainable AI (XAI) = AI where humans can understand the reasoning behind decisions/predictions. XAI algorithms follow three principles:
\begin{itemize}
    \item \textbf{Transparency}: the processes extracting model parameters from training data and generating labels from test data can be described by the designer.
    \item \textbf{Interpretability}: the possibility of comprehending the ML model and presenting the underlying basis for decision-making in a human-understandable way.
    \item \textbf{Explainability}: the collection of features of the interpretable domain that contributed to a given prediction (e.g., classification or regression).
\end{itemize}

ML models can be categorized as \textbf{white-box} or \textbf{black-box}:
\begin{itemize}
    \item \textbf{White-box}: results are understandable to domain experts. E.g., linear/logistic regression, kNN, linear SVM, decision tree, GLMs, GAMs, k-means, PCA.
    \item \textbf{Black-box}: extremely hard to explain even for domain experts. E.g., ensemble learning (XGBoost, LightGBM), kernel methods, neural networks (MLP, CNN, RNN, Transformer, GNN).
\end{itemize}

\textbf{Key Definitions:}
\begin{itemize}
    \item A \textbf{model explanation} attempts to highlight why a model made a prediction.
    \item A \textbf{feature importance explanation} focuses on each feature's role:
    \begin{itemize}
        \item \textit{Feature attribution}: each feature $x_i$ receives a score $a_i \in \mathbb{R}$.
        \item \textit{Feature selection}: a subset of important features $x_S \in \{x_1, \ldots, x_d\}$.
    \end{itemize}
    \item Explanations may be \textbf{local} (individual prediction) or \textbf{global} (overall model behavior).
    \item An \textbf{explanation algorithm} generates explanations given input data and an ML model.
\end{itemize}

\subsection{Permutation Test}

\textbf{Permutation test} (a.k.a.\ permutation feature importance) determines \textbf{global} importance of each feature:
\begin{enumerate}
    \item Evaluate model accuracy on original data.
    \item For each feature $x_i$: corrupt it by randomly shuffling that column, then record accuracy drop:
    \[
        a_i = \mathrm{Acc}(\text{original}) - \mathrm{Acc}(x_i\ \text{corrupted})
    \]
\end{enumerate}

More precisely, the importance score is:
\[
    a_i = \frac{1}{n}\sum_{j=1}^{n} \ell\!\left(f\!\left(x_1^j,\ldots,\tilde{x}_i^j,\ldots,x_d^j\right), y^j\right)
    - \frac{1}{n}\sum_{j=1}^{n} \ell\!\left(f\!\left(x_1^j,\ldots,x_i^j,\ldots,x_d^j\right), y^j\right)
\]
where $\ell(\cdot,\cdot)$ is an arbitrary loss function and $\tilde{x}_i^j$ denotes the corrupted (permuted) $i$-th feature.

\textbf{Properties:}
\begin{itemize}
    \item Model-agnostic; supports continuous/categorical features.
    \item Fast and easy to implement.
    \item Empirical, no theoretical guarantees.
    \item Problematic for correlated features.
\end{itemize}

\subsection{Occlusion}

\textbf{Occlusion} explains \textbf{local} predictions for image classifiers by masking regions and measuring the prediction change. It calculates pixel (or super-pixel) importance.

\textbf{Process:}
\begin{enumerate}
    \item Make prediction on the full image.
    \item Occlude each region (replace with zero/uninformative pixels; can use 2$\times$2, 4$\times$4 patches, or super-pixels), record how the prediction changes.
\end{enumerate}

Mathematically, the importance score of feature $x_i$ is:
\[
    a_i = f_y(x_1, \ldots, x_i, \ldots, x_d) - f_y(x_1, \ldots, 0, \ldots, x_d)
\]

\textbf{Properties:}
\begin{itemize}
    \item Works with any model, even non-image data.
    \item Moderately fast: $d+1$ model evaluations per prediction.
    \item Easy to implement; empirical, no theoretical guarantees.
\end{itemize}

\subsection{LIME}

\textbf{LIME} (Local Interpretable Model-agnostic Explanations): feature attribution for an individual sample $\mathbf{x}$.

\textbf{Setup:} Let $\mathbf{x} \in \mathbb{R}^d$ be the sample, $f: \mathbb{R}^d \to \mathbb{R}$ the prediction model, and $\mathbf{x}' \in \{0,1\}^{d'}$ its binary interpretable representation (e.g., patch indicator vector for images). LIME solves:
\[
    \xi(\mathbf{x}) = \underset{g \in G}{\mathrm{argmin}}\ \mathcal{L}(f, g, \pi_x) + \Omega(g)
\]
where:
\begin{itemize}
    \item $g: \{0,1\}^{d'} \to \mathbb{R}$: interpretable model, e.g., $g(z') = \mathbf{w}^\top \mathbf{z}'$.
    \item $h_x: \{0,1\}^{d'} \to \mathbb{R}^d$: maps $\mathbf{z}'$ back to the original space of $\mathbf{x}$, yielding $\mathbf{z}$.
    \item $\pi_x(z)$: proximity measure between $\mathbf{z}$ and $\mathbf{x}$, e.g., $\exp(-\|\mathbf{x} - \mathbf{z}\|^2/\sigma^2)$.
    \item $\mathcal{L}(f, g, \pi_x)$: fidelity loss measuring how unfaithful $g$ approximates $f$ in the locality of $\pi_x$:
    \[
        \mathcal{L}(f, g, \pi_x) = \sum_{(\mathbf{z}, \mathbf{z}')} \pi_x(\mathbf{z})\bigl(f(\mathbf{z}) - g(\mathbf{z}')\bigr)^2
    \]
    \item $\Omega(g)$: complexity measure of $g$.
\end{itemize}

\textbf{Main Steps:}
\begin{enumerate}
    \item Given $\mathbf{x}$, form $\mathbf{x}'$.
    \item Sample $\mathbf{z}'$ around $\mathbf{x}'$ by drawing nonzero elements uniformly at random.
    \item Map each $\mathbf{z}'$ to the original space via $h_x$ to obtain $\mathbf{z}$.
    \item Solve the weighted regression (e.g., with Lasso):
    \[
        \min_{\mathbf{w}}\ \sum_{(\mathbf{z},\mathbf{z}')} \exp\!\left(-\frac{\|\mathbf{x}-\mathbf{z}\|^2}{\sigma^2}\right)\bigl(f(\mathbf{z}) - \mathbf{w}^\top \mathbf{z}'\bigr)^2 + \lambda\|\mathbf{w}\|_1
    \]
    \item Use $\mathbf{w}$ to explain: a larger $w_i$ means feature $i$ contributes more to the positive prediction.
\end{enumerate}

\subsection{SHAP}

\textbf{SHAP} (SHapley Additive exPlanations): based on Shapley values from cooperative game theory (1953). Each feature value is treated as a "player" in a game where the model prediction is the payout. SHAP unifies various existing XAI methods with axiomatic definitions.

\textbf{Shapley Values Setup:}
\begin{itemize}
    \item Set of players $D = \{1, \ldots, d\}$; a coalition is any $S \subseteq D$.
    \item A game is represented by a characteristic function $v: 2^D \to \mathbb{R}$, assigning a value to every coalition.
    \item Special cases: grand coalition $v(D)$, null coalition $v(\varnothing)$, arbitrary coalition $v(S)$.
    \item Shapley values allocate credit fairly to players, derived from fairness axioms.
\end{itemize}

The Shapley value is a function $\phi: G \to \mathbb{R}^d$, assigning one credit per player. For a game $v$, the Shapley values are $\phi_1(v), \ldots, \phi_d(v)$, computed as the weighted average marginal contribution over all coalitions $S \subseteq D \setminus \{i\}$:
\[
    \phi_i(v) = \sum_{S \subseteq D \setminus \{i\}} \frac{|S|!\,(d-1-|S|)!}{d!}\bigl[v(S \cup \{i\}) - v(S)\bigr]
\]

\textbf{Axiomatic uniqueness:} The Shapley value is the \emph{only} attribution method $\phi$ satisfying all four axioms:
\begin{itemize}
    \item \textbf{Efficiency}: $\sum_{i \in D} \phi_i(v) = v(D) - v(\varnothing)$.
    \item \textbf{Symmetry}: If $v(S \cup \{i\}) = v(S \cup \{j\})$ for all $S \subseteq D$, then $\phi_i(v) = \phi_j(v)$.
    \item \textbf{Null player}: If $v(S \cup \{i\}) = v(S)$ for all $S \subseteq D$, then $\phi_i(v) = 0$.
    \item \textbf{Linearity}: $\phi(c_1 v_1 + c_2 v_2) = c_1\phi(v_1) + c_2\phi(v_2)$, where $c_1, c_2 \in \mathbb{R}$.
\end{itemize}

\textbf{SHAP for ML models:} Features are players, model prediction is the payout. The characteristic function is replaced by the ML model retrained on each feature subset $S \subseteq D$:
\[
    \phi_i = \sum_{S \subseteq D \setminus \{i\}} \frac{|S|!\,(d-|S|-1)!}{d!}\bigl[f_{S \cup \{i\}}(x_{S \cup \{i\}}) - f_S(x_S)\bigr]
\]
Requires retraining the model on all feature subsets $S \subseteq D$.

\textbf{Computation Complexity:} Exact computation is $O(2^d)$ — intractable for $d > 20$. Approximation methods exist (e.g., Kernel SHAP, Deep SHAP).

\textbf{SHAP Methods:}
\begin{itemize}
    \item \textbf{Kernel SHAP}: model-agnostic approximation.
    \item Max SHAP, Deep SHAP: model-type-specific.
\end{itemize}

\subsubsection{Kernel SHAP}

Kernel SHAP = Linear LIME + Shapley values. Let $g(\mathbf{z}') = w_0 + \sum_{i=1}^d w_i z_i'$, $\mathbf{z}' \in \{0,1\}^{d'}$. Kernel SHAP solves:
\[
    \min_{w_0,\ldots,w_d}\ \sum_{\mathbf{z}'_S} \pi(S)\!\left(w_0 + \sum_{i \in S} w_i - f(h_x(\mathbf{z}'_S))\right)^2, \quad
    \pi(S) = \frac{d-1}{\binom{d}{|S|}|S|(d-|S|)}
\]

This simplifies to $\min_{\mathbf{w}} \|\mathbf{y}_\pi - \mathbf{U}_\pi \mathbf{w}\|^2$, with closed-form solution $(\mathbf{U}_\pi^\top \mathbf{U}_\pi)^{-1}\mathbf{U}_\pi^\top \mathbf{y}_\pi$. It can be proved that:
\[
    w_i = \phi_i = \sum_{S \subseteq D \setminus \{i\}} \frac{|S|!(d-|S|-1)!}{d!}\bigl[f(h_x(z'_{S\cup\{i\}})) - f(h_x(z'_S))\bigr]
\]

\subsection{XAI Summary}

Removal-based methods are common in XAI (Permutation Test, Occlusion, LIME, SHAP). Other structural approaches:
\begin{itemize}
    \item \textbf{Gradient-based} (e.g., Grad-CAM, Gradient$\times$Input): use the gradient of network output w.r.t.\ each input variable.
    \item \textbf{Propagation-based} (e.g., Deep Taylor, Deep SHAP): use backpropagation to estimate attributions at each layer and propagate them to the previous layer.
\end{itemize}

\subsection{Interpretable ML vs XAI}

\begin{itemize}
    \item \textbf{Interpretable ML}: use a white-box model directly (not a black-box).
    \item \textbf{XAI}: use a black-box model, then use another model to explain it --- start with a black-box, create a proxy model that approximates it, compute derivatives or visualize attention.
    \item \textbf{Interpretable ML $\neq$ XAI}: XAI introduces ``Double Trouble'' --- relying on two models instead of one, both of which may make mistakes and disagree. If an interpretable model suffices, there is no reason to explain a black-box. (Example: COMPAS vs.\ CORELS)
\end{itemize}
